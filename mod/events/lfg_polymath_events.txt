# LFG Traditions - Polymath Commons events.
namespace = lfg_polymath

# Yearly (on_yearly_pulse_country): with Reckless Experimentation, 5% chance of a lab accident and 15% chance of a
# lab breakthrough on a random colony.
country_event = {
	id = lfg_polymath.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_tradition = tr_lfg_polymath_reckless_experimentation
		any_owned_planet = { is_capital = no }
	}

	immediate = {
		random_list = {
			80 = { }
			5 = {
				random_owned_planet = {
					limit = { is_capital = no }
					add_modifier = { modifier = lfg_lab_accident days = 1800 }
					save_event_target_as = lfg_lab_accident_planet
				}
				country_event = { id = lfg_polymath.2 }
			}
			15 = {
				random_owned_planet = {
					limit = { is_capital = no }
					add_modifier = { modifier = lfg_lab_breakthrough days = 1800 }
					save_event_target_as = lfg_lab_breakthrough_planet
				}
				country_event = { id = lfg_polymath.3 }
			}
		}
	}
}

# Notification of the accident.
country_event = {
	id = lfg_polymath.2
	title = lfg_polymath.2.name
	desc = lfg_polymath.2.desc
	picture = GFX_evt_large_explosion
	is_triggered_only = yes

	option = {
		name = lfg_polymath.2.a
	}
}
