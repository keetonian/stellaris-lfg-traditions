# LFG Traditions - Sealed Ring origin (origin_lfg_sealed_ring): an intact ring world beyond the galactic rim.
namespace = lfg_sealed_ring

# Game start (on_game_start_country):
# - no starting ships (lfg_sealed_ring_remove_starting_fleets; stations stay);
# - remove vanilla's homeworld blockers (Industrial Wasteland, Sprawling Slums...), set the homeworld's districts
#   (lfg_sealed_ring_starting_districts), then seal the ring's segments (d_lfg_sealed_section: 3 on the homeworld, 7
#   on each other habitable segment), give the homeworld Gigastructural's Great Ring mining node, and put +10 minerals
#   and the Quasi-Negative Mass deposit (with a research station) on the two Tiyanki Residue bodies;
# - give Negative Energy Studies (QNM only produces with it; the Urmazin Traders, met at the end of the Whale Road,
#   buy it);
# - move the home system beyond the rim, like Gigastructural's Away On An Island origin (ehof_incohesive.3): a random
#   replacement system takes over its hyperlanes, then the system is moved 140-170 out from a rim system until
#   lfg_sealed_ring_isolated holds (no galaxy system within 130); then spawn its one neighbour, 15-25 away, with a
#   hyperlane. A plain jump drive (range 100) can't reach the galaxy from there; with Sealed Ring Drive (250) it can.
country_event = {
	id = lfg_sealed_ring.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_lfg_sealed_ring
		exists = capital_scope
	}

	immediate = {
		set_country_flag = lfg_sealed_ring_origin
		give_technology = { tech = tech_negative_e_s message = no }

		lfg_sealed_ring_remove_starting_fleets = yes
		country_event = { id = lfg_sealed_ring.5 days = 1 }
		capital_scope = {
			# Vanilla's homeworld blockers (game_start.12, at empire creation, before this) go; capital_scope is the
			# colony, which holds a settled planet's deposits.
			clear_blockers = yes
			lfg_sealed_ring_starting_districts = yes
			while = {
				count = 3
				add_deposit = d_lfg_sealed_section
			}
			# Mining jobs, as Gigastructural's Great Ring homeworld gets them: its Ringworld Self-Repair System Node
			# (5 Matter Synthesizer jobs: miners, 4 minerals each).
			add_deposit = d_great_ring_excavated_mountain
		}
		capital_scope.solar_system = {
			save_event_target_as = lfg_sealed_ring_home
			# The other three habitable segments (flagged in lfg_sealed_ring_start: capital_scope is a colony, so
			# comparing planets with it never matched and the homeworld got these too).
			every_system_planet = {
				limit = { has_planet_flag = lfg_sealed_ring_segment }
				clear_blockers = yes
				while = {
					count = 7
					add_deposit = d_lfg_sealed_section
				}
			}
			# +10 minerals on the second Tiyanki Residue body, for a mining station.
			every_system_planet = {
				limit = { has_planet_flag = lfg_sealed_ring_minerals }
				clear_deposits = yes
				add_deposit = d_minerals_10
			}
			every_system_planet = {
				limit = { has_planet_flag = lfg_sealed_ring_negative_mass }
				clear_deposits = yes
				add_deposit = d_lfg_sealed_ring_negative_mass
				create_research_station = { owner = root }
			}
		}

		# Hand the home system's hyperlanes to a replacement system in its place.
		event_target:lfg_sealed_ring_home = {
			spawn_system = {
				initializer = random
				min_distance >= 0
				max_distance <= 1
				hyperlane = no
			}
			last_created_system = { save_event_target_as = lfg_sealed_ring_replacement }
			every_neighbor_system = {
				set_star_flag = lfg_sealed_ring_old_neighbor
				add_hyperlane = { from = this to = event_target:lfg_sealed_ring_replacement }
			}
		}
		every_system = {
			limit = { has_star_flag = lfg_sealed_ring_old_neighbor }
			remove_hyperlane = { from = this to = event_target:lfg_sealed_ring_home }
			remove_star_flag = lfg_sealed_ring_old_neighbor
		}

		# Beyond the rim.
		event_target:lfg_sealed_ring_home = {
			while = {
				count = 100
				if = {
					limit = { NOT = { has_star_flag = lfg_sealed_ring_placed } }
					random_system = {
						limit = {
							lfg_sealed_ring_galaxy_system = yes
							distance_to_core_percent >= 85
						}
						save_event_target_as = lfg_sealed_ring_anchor
					}
					move_system = {
						target = event_target:lfg_sealed_ring_anchor
						min_distance >= 140
						max_distance <= 170
						hyperlane = no
						reset_terra_incognita = yes
					}
					if = {
						limit = { lfg_sealed_ring_isolated = yes }
						set_star_flag = lfg_sealed_ring_placed
					}
				}
			}
			if = {
				limit = { NOT = { has_star_flag = lfg_sealed_ring_placed } }
				log = "lfg_sealed_ring.1: no isolated spot beyond the rim found; the home system stays at its last try"
			}

			set_spawn_system_batch = begin
			spawn_system = {
				min_distance >= 15
				max_distance <= 25
				initializer = lfg_sealed_ring_neighbor_init
				hyperlane = no
			}
			set_spawn_system_batch = end
		}
		random_system = {
			limit = { has_star_flag = lfg_sealed_ring_neighbor }
			save_event_target_as = lfg_sealed_ring_neighbor_system
			add_hyperlane = { from = this to = event_target:lfg_sealed_ring_home }
		}

		# The Matriarch's herd in the neighbour: two families of vanilla's garrison Tiyanki (they don't roam), placed like
		# vanilla's displaced hatchlings (anomaly_events_cosmic_storms). We start in contact with the Tiyanki, as
		# vanilla's Fruitful Partnership does (fruitful_start_effect: the tiyanki_encountered flag + create_tiyanki_country).
		create_tiyanki_garrison_country = yes
		set_country_flag = tiyanki_encountered
		create_tiyanki_country = yes
		establish_communications_no_message = event_target:tiyanki_country
		establish_communications_no_message = event_target:tiyanki_garrison_country
		event_target:lfg_sealed_ring_neighbor_system = {
			log = "lfg_sealed_ring.1: neighbour system [This.GetName]"
			random_system_planet = {
				limit = { is_star = yes }
				save_event_target_as = lfg_sealed_ring_herd_point
			}
		}
		while = {
			count = 2
			create_fleet = {
				name = "NAME_Tiyanki_Space_Whale"
				effect = {
					set_owner = event_target:tiyanki_garrison_country
					create_ship = { design = "NAME_Tiyanki_Cow" }
					create_ship = { design = "NAME_Tiyanki_Bull" }
					create_ship = { design = "NAME_Tiyanki_Calf" }
					create_ship = { design = "NAME_Tiyanki_Hatchling" }
					set_location = {
						target = event_target:lfg_sealed_ring_herd_point
						distance = 40
						angle = random
					}
					set_fleet_stance = passive
					set_aggro_range_measure_from = self
					set_aggro_range = 0
					log = "lfg_sealed_ring.1: Tiyanki herd [This.GetName] in [This.Solar_System.GetName]"
				}
			}
		}

		# The Whale Road: a long dig site on one of the neighbour's molten worlds; its end opens a wormhole to the
		# Matriarch (lfg_whale_road_digsite, lfg_sealed_ring.15).
		event_target:lfg_sealed_ring_neighbor_system = {
			random_system_planet = {
				limit = {
					is_planet_class = pc_molten
					is_moon = no
				}
				create_archaeological_site = lfg_whale_road_digsite
				log = "lfg_sealed_ring.1: Whale Road dig site on [This.GetName]"
			}
		}

		# Only one Legendary Matriarch, however many empires have the origin: beside the first one's ring.
		if = {
			limit = { NOT = { has_global_flag = lfg_legendary_matriarch_spawned } }
			set_global_flag = lfg_legendary_matriarch_spawned
			# The Matriarch: her own system, moved 140-170 beyond a random rim system (as the ring is) until it's far from every
			# Swallowed Whole ring (lfg_sealed_ring_matriarch_isolated): somewhere else around the galaxy.
			event_target:lfg_sealed_ring_home = {
				set_spawn_system_batch = begin
				spawn_system = {
					min_distance >= 50
					max_distance <= 80
					initializer = lfg_sealed_ring_matriarch_init
					hyperlane = no
				}
				set_spawn_system_batch = end
			}
			random_system = {
				limit = {
					has_star_flag = lfg_sealed_ring_matriarch
					NOT = { has_star_flag = lfg_sealed_ring_placed }
				}
				save_event_target_as = lfg_sealed_ring_matriarch_system
				while = {
					count = 100
					if = {
						limit = { NOT = { has_star_flag = lfg_sealed_ring_placed } }
						if = {
							limit = { lfg_sealed_ring_matriarch_isolated = yes }
							set_star_flag = lfg_sealed_ring_placed
						}
						else = {
							random_system = {
								limit = {
									lfg_sealed_ring_galaxy_system = yes
									distance_to_core_percent >= 85
								}
								save_event_target_as = lfg_sealed_ring_matriarch_anchor
							}
							move_system = {
								target = event_target:lfg_sealed_ring_matriarch_anchor
								min_distance >= 140
								max_distance <= 170
								hyperlane = no
								reset_terra_incognita = yes
							}
						}
					}
				}
				if = {
					limit = { NOT = { has_star_flag = lfg_sealed_ring_placed } }
					log = "lfg_sealed_ring.1: no spot far from the ring found for the Matriarch; she stays at its last try"
				}
			}
			# Vanilla's ancient Tiyanki (guardian_elderly_tiyanki, its power, and a bigger copy of its Progenitor design:
			# GENERATED lfg_matriarch_* files) made legendary (lfg_legendary_matriarch). The country flag marks her for lfg_sealed_ring.4.
			create_country = {
				name = "NAME_lfg_legendary_matriarch"
				type = guardian_elderly_tiyanki
				flag = {
					icon = { category = "zoological" file = "flag_zoological_17.dds" }
					background = { category = "backgrounds" file = "00_solid.dds" }
					colors = { "dark_purple" "dark_purple" "null" "null" }
				}
			}
			last_created_country = {
				set_country_flag = lfg_legendary_matriarch_country
				add_modifier = { modifier = elderly_tiyanki_power days = -1 }
				add_modifier = { modifier = lfg_legendary_matriarch days = -1 }
				save_event_target_as = lfg_legendary_matriarch_country
			}
			event_target:lfg_sealed_ring_matriarch_system = {
				random_system_planet = {
					limit = { is_star = yes }
					save_event_target_as = lfg_legendary_matriarch_start
				}
			}
			event_target:lfg_legendary_matriarch_country = {
				create_fleet = {
					name = "NAME_lfg_legendary_matriarch"
					settings = {
						spawn_debris = no
						is_boss = yes
					}
					effect = {
						set_owner = event_target:lfg_legendary_matriarch_country
						create_ship = {
							name = "NAME_lfg_legendary_matriarch"
							design = "NAME_lfg_legendary_matriarch"	# GENERATED: vanilla's Progenitor, model x1.75
						}
						set_location = {
							target = event_target:lfg_legendary_matriarch_start
							distance = 50
							angle = random
						}
					log = "lfg_sealed_ring.1: Legendary Matriarch in [This.Solar_System.GetName]"
						set_fleet_stance = aggressive
						set_aggro_range_measure_from = self
						set_aggro_range = 50
					}
				}
			}
		}
	}
}

# on_ship_destroyed_perp (this = the killer's owner, from = the destroyed ship's owner, fromfromfrom = that ship): the
# Legendary Matriarch is dead. Her killer may take her DNA: vanilla's harvested_tiyanki_dna flag, which makes
# Polymelic (trait_tiyanki) available to species with leviathan genes (Biomorphosis mutation), as killing vanilla's
# Matriarch does (distar.212).
country_event = {
	id = lfg_sealed_ring.4
	title = lfg_sealed_ring.4.name
	desc = lfg_sealed_ring.4.desc
	picture = GFX_evt_ancient_tiyanki
	show_sound = event_space_whale
	is_triggered_only = yes

	trigger = {
		is_country_type = default
		from = { has_country_flag = lfg_legendary_matriarch_country }
		NOT = { has_country_flag = lfg_legendary_matriarch_slain }
	}

	immediate = {
		set_country_flag = lfg_legendary_matriarch_slain
		set_country_flag = harvested_tiyanki_dna
		fromfromfrom = {
			create_ambient_object = {
				type = "elderly_tiyanki_dead_object"
				location = this
				effect = { save_global_event_target_as = lfg_dead_legendary_matriarch }
			}
		}
	}

	option = {
		name = lfg_sealed_ring.4.a
		custom_tooltip = lfg_sealed_ring.4.a.tt
	}

	# Reanimators can raise her (as vanilla's distar.212 offers for its Matriarch): a special project on her corpse.
	# is_reanimator covers Reanimated Armies, Permanent Employment, Cordyceptic Drones (and Cordyceptic Symbionts, its
	# Wilderness swap_type: same civic key) and Mechromancy.
	option = {
		name = lfg_sealed_ring.4.b
		trigger = { is_reanimator = yes }
		custom_tooltip = lfg_sealed_ring.4.a.tt
		event_target:lfg_dead_legendary_matriarch = {
			enable_special_project = {
				name = "LFG_REANIMATE_LEGENDARY_MATRIARCH"
				location = this
				owner = root
			}
		}
	}
}

# Stage 6: the end of the Whale Road. Contact with the Urmazin Traders, and a natural wormhole pair links the dig site's system and the Matriarch's (as
# vanilla's distar.290 links the Sealed System); natural wormholes need Wormhole Stabilization: 25% of it (and the
# research option) if missing.
fleet_event = {
	id = lfg_sealed_ring.15
	title = lfg_sealed_ring.15.name
	desc = lfg_sealed_ring.15.desc
	picture = GFX_evt_ancient_tiyanki
	show_sound = event_mystic_reveal
	location = from
	is_triggered_only = yes
	archaeology = yes

	immediate = {
		from = { set_site_progress_locked = yes }
	}
	after = {
		from = { set_site_progress_locked = no }
	}

	option = {
		name = lfg_sealed_ring.15.a
		medium_artifact_reward = yes
		custom_tooltip = lfg_sealed_ring.15.urmazin_tt
		# The Urmazin Traders: spawned now if they aren't there yet (ehof_mega_trader.100, which also gives every regular
		# empire communications with them, as in Gigastructural's own story), in contact with us, their first-visit
		# greeting skipped so we can trade from the diplomacy screen at once.
		hidden_effect = {
			owner = {
				if = {
					limit = { NOT = { has_global_flag = ehof_traders_spawned } }
					if = {
						limit = { NOT = { exists = event_target:current_system_creator } }
						save_global_event_target_as = current_system_creator
					}
					country_event = { id = ehof_mega_trader.100 }
				}
				set_country_flag = ehof_traders_first_time
				country_event = { id = lfg_sealed_ring.8 days = 1 }
			}
		}
		owner = {
			add_monthly_resource_mult = {
				resource = physics_research
				value = 24
				min = 600
				max = 6000
			}
		}
		custom_tooltip = lfg_sealed_ring.15.a.tt
		owner = {
			if = {
				limit = { NOT = { has_technology = tech_wormhole_stabilization } }
				add_research_option = tech_wormhole_stabilization
				add_tech_progress = { tech = tech_wormhole_stabilization progress = 0.25 }
			}
		}
		hidden_effect = {
			from.solar_system = { save_event_target_as = lfg_whale_road_entry }
			random_system = {
				limit = { has_star_flag = lfg_sealed_ring_matriarch }
				save_event_target_as = lfg_whale_road_exit
			}
			if = {
				limit = { exists = event_target:lfg_whale_road_exit }
				event_target:lfg_whale_road_entry = {
					event_target:lfg_whale_road_exit = {
						prev = {
							spawn_natural_wormhole = {
								bypass_type = wormhole
								random_pos = no
								orbit_angle = 360
							}
						}
						spawn_natural_wormhole = {
							bypass_type = wormhole
							random_pos = no
							orbit_angle = 270
						}
						link_wormholes = prev
					}
				}
			}
			# Natural wormholes must be "seen" (bypass type wormhole: remember_instances) before ships can use them; ours
