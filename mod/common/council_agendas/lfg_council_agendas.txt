# LFG Traditions - council agendas, unlocked by tradition finishers (`unlocks_agenda`).
# `modifier` applies while the council prepares the agenda; on launch `effect` runs once and the
# `finish_modifier` (common/static_modifiers/lfg_static_modifiers.txt) lasts 10 years (COUNCIL_AGENDA_FINISH_MODIFIER_DURATION).

# Two versions, like vanilla's agenda_conquer_nature / _nomads: only one is ever available, depending on
# whether Gigastructural's supertensiles exist in this game (global flag giga_buildcap_j). Unlocked by adopting
# the tree (tr_lfg_megaforge_adopt); its unlocks_agenda names the supertensile version; the classic one is offered through its own allow/potential.
agenda_lfg_great_project = {
	agenda_cost = @lfg_agenda_cost
	allow = {
		has_tradition = tr_lfg_megaforge_adopt
	}
	potential = {
		has_paragon_dlc = yes
		is_nomadic = no
		has_global_flag = giga_buildcap_j
	}
	modifier = {
		megastructure_build_speed_mult = 0.05
	}
	finish_modifier = agenda_lfg_great_project_finish
}

agenda_lfg_great_project_classic = {

agenda_lfg_harvest_tithe = {
	agenda_cost = @lfg_agenda_cost
	allow = {
		has_tradition = tr_lfg_abundance_finish
	}
	potential = {
		has_paragon_dlc = yes
		lfg_main_species_eats_food = yes
	}
	modifier = {
		planet_farmers_produces_mult = 0.05
	}
	finish_modifier = agenda_lfg_harvest_tithe_finish
	effect = {
		if = {
			limit = { resource_stockpile_compare = { resource = food value >= 3000 } }
			add_resource = {
				food = -3000
				influence = 150
				unity = 2000
			}
		}
	}
}
