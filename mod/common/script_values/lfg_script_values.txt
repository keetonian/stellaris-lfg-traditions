# LFG Traditions - script values.

# Extra monthly progress for the Synthetic Fertility situation (situation_synthetic_fertility_digitization,
# see common/situations/zz_lfg_situation_overrides.txt): +1 per year since game start, which is when the
# origin's situation begins. Year 10 => +10 per month on top of the vanilla base of 15.
lfg_synthetic_fertility_acceleration = {
	base = 0
	add = trigger:years_passed
	mult = 1
}

# Share of what's left of the source species that the next split-off subspecies takes (see lfg_split_species).
# root is the country running the split; lfg_split_remaining counts down 50, 49, ... 1.
lfg_species_split_share = {
	base = 1
	divide = root.lfg_split_remaining
}

#####################################################################
# Tradition scaling. Used as `mult = value:<name>` in triggered_modifiers; each is capped with `max`.
#####################################################################

# Polymath Commons - Many Minds: species in the empire (genetic variants don't count), max 30.
lfg_num_species = {
	base = 0
	complex_trigger_modifier = {
		trigger = num_species
		mode = add
	}
	max = 30
}

# Polymath Commons - Rival Laboratories / Galactic Broadcast - Ratings War: current rivals, max 4.
lfg_num_rivals = {
	base = 0
	complex_trigger_modifier = {
		trigger = num_rivals
		mode = add
	}
	max = 4
}

# Galactic Broadcast: empires we have communications with, max 20.
lfg_num_contacts = {
	base = 0
	complex_trigger_modifier = {
		trigger = num_communications
		mode = add
	}
	max = 20
}

# Granary Kings - Bumper Crops (and its Feast of Plenty doubling): monthly food income (net, as vanilla's
# has_monthly_income reads it) in units of 1,000, never negative. Pattern from vanilla income_mult (01_script_values_paragon.txt).
lfg_food_income_1000s = {
	base = 0
	complex_trigger_modifier = {
		trigger = has_monthly_income
		parameters = {
			resource = food
		}
		mode = add
	}
	divide = 1000
	min = 0
}

# Megaforge Doctrine - Monument Economy: megastructures owned, including Gigastructural kilostructures but not
# ruined (unrepaired) megastructures (Giga's is_ruined_mega, giga_mega_categories.txt). Uncapped.
lfg_num_megastructures = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_megastructure
		parameters = {
			limit = {
				is_ruined_mega = no
			}
		}
		mode = add
	}
}

# The Grand Tour: hundreds of Resort Workers employed, uncapped. An empire can only have one
# resort world (vanilla decision_resort_colony_only_one), so this measures how built-up it is.
# Pattern from vanilla num_active_haruspex (04_script_values_machine_age.txt).
lfg_resort_worker_hundreds = {
	base = 0
	complex_trigger_modifier = {
		trigger = num_assigned_jobs
		parameters = {
			job = resort_worker
		}
		mode = add
	}
	divide = 100
}

# The Grand Tour - Galactic Destination: tourists (civilians) on the resort world / 1000, uncapped. Country variable
# lfg_tourists is refreshed monthly by lfg_resorts.1.
lfg_tourist_thousands = {
	base = 0
	complex_trigger_modifier = {
		trigger = check_variable
		parameters = {
			which = lfg_tourists
		}
		mode = add
	}
	divide = 1000
}

# The Grand Tour - Expand the Resort decision cost multiplier (planet scope): 1, 2, 3, 4, 5 for the
# 1st..5th expansion, i.e. 1 + expansions already made (planet variable lfg_resort_expansions).
lfg_resort_expansion_cost_mult = {
	base = 1
	complex_trigger_modifier = {
		trigger = check_variable
		parameters = {
			which = lfg_resort_expansions
		}
		mode = add
	}
}
