# LFG Traditions - policies.

# Monument Economy (Megaforge Doctrine slot 5): choose which side of the economy each megastructure feeds.
# The policy only sets a flag; the scaling bonuses live in tr_lfg_megaforge_monuments (common/traditions), which
# reads the flag and multiplies by value:lfg_num_megastructures. The first option (Balanced, no effect) is the default.
lfg_monument_economy = {
	category = economy
	potential = {
		has_tradition = tr_lfg_megaforge_monuments
	}

	option = {
		name = "lfg_monument_economy_balanced"
		policy_flags = {}
	}

	option = {
		name = "lfg_monument_economy_monuments_first"
		policy_flags = {
			lfg_monuments_first
		}
	}

	option = {
		name = "lfg_monument_economy_people_first"
		policy_flags = {
			lfg_people_first
		}
	}
}
