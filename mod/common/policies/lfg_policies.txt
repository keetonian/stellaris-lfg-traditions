# LFG Traditions - policies.

# Monument Economy (Megaforge slot 5): choose which side of the economy each megastructure feeds.
# The policy only sets a flag; the scaling bonuses live in tr_lfg_megaforge_monuments (common/traditions), which
# reads the flag and multiplies by value:lfg_num_megastructures. The first option (Balanced, no effect) is the default.
lfg_monument_economy = {
	category = economy
	potential = {
		has_tradition = tr_lfg_megaforge_monuments
	}

	option = {
		name = "lfg_monument_economy_balanced"
		policy_flags = {}
		ai_weight = {
			weight = 1
		}
	}

	option = {
		name = "lfg_monument_economy_monuments_first"
		policy_flags = {
			lfg_monuments_first
		}
		ai_weight = {
			weight = 1
		}
	}

	option = {
		name = "lfg_monument_economy_people_first"
		policy_flags = {
			lfg_people_first
		}
		ai_weight = {
			weight = 1
		}
	}
}

# Nanite Forge origin: turn part of our nanite production into unity or research (lfg_nanite_forge.7 reads the flag
# monthly; value:lfg_nanite_production). The first option (no conversion) is the default.
lfg_nanite_conversion = {
	category = economy
	potential = {
		has_origin = origin_lfg_nanite_forge
	}

	option = {
		name = "lfg_nanite_conversion_none"
		policy_flags = {}
		ai_weight = {
			weight = 1
		}
	}

	option = {
		name = "lfg_nanite_conversion_unity"
		policy_flags = {
			lfg_nanite_conversion_unity
		}
		ai_weight = {
			weight = 2
			modifier = {
				factor = 0
				has_deficit = nanites
			}
		}
	}

	option = {
		name = "lfg_nanite_conversion_research"
		policy_flags = {
			lfg_nanite_conversion_research
		}
		ai_weight = {
			weight = 3
			modifier = {
				factor = 0
				has_deficit = nanites
			}
		}
	}
}
