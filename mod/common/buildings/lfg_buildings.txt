# LFG Traditions - buildings (new ones; edits to vanilla buildings are generated in zz_lfg_building_overrides.txt).
# The two nanite automation buildings cost half (build cost and upkeep) on Nanite Worlds (pc_nanotech): cost blocks
# with mult = @lfg_nanite_world_discount, and their upkeep values (lfg_resource_districts / lfg_all_districts).

# Nanite Forge origin: like vanilla's Technician / Miner / Farmer automation buildings (01_pop_assembly_buildings.txt),
# but one building for all three jobs (and their drone versions), in any of their district specializations. Upkeep is
# nanites per resource district on the planet (value:lfg_resource_districts). Automated jobs still pay vanilla's own
# automation upkeep.
building_lfg_nanite_automation = {
	base_buildtime = @b2_time
	planet_limit = 1
	category = pop_assembly
	icon = lfg_nanite_automation	# art/make_building_icons.py

	building_sets = {
		generator_automation
		mining_automation
		farming_automation
	}

	potential = {
		exists = owner
		owner = { has_origin = origin_lfg_nanite_forge }
	}

	planet_modifier = {
		job_technician_automated_workforce_mult = @lfg_nanite_hub_automation
		job_technician_drone_automated_workforce_mult = @lfg_nanite_hub_automation
		job_miner_automated_workforce_mult = @lfg_nanite_hub_automation
		job_mining_drone_automated_workforce_mult = @lfg_nanite_hub_automation
		job_farmer_automated_workforce_mult = @lfg_nanite_hub_automation
		job_agri_drone_automated_workforce_mult = @lfg_nanite_hub_automation
	}

	resources = {
		category = planet_buildings
		cost = {
			trigger = { NOT = { is_planet_class = pc_nanotech } }
			minerals = @b2_minerals
			nanites = 100
		}
		cost = {
			trigger = { is_planet_class = pc_nanotech }
			minerals = @b2_minerals
			nanites = 100
			mult = @lfg_nanite_world_discount
		}
		upkeep = {
			nanites = @lfg_nanite_hub_upkeep_per_district
			mult = value:lfg_resource_districts
		}
	}
}
