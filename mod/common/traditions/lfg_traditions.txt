# LFG Traditions - traditions. Layout per tree (tree_21_11): slot 4 requires slots 1 + 2, slot 5 requires slot 3.
#
# Every tree has a "hook": bonuses that scale with or depend on the game state (stockpiles, rivals, contacts,
# species, megastructures, artificial worlds, resort workers, war/peace), so they change how you play.
# Scaling bonuses use `triggered_modifier = { ... mult = value:<script value> }` (common/script_values) and are explained
# with a custom_tooltip, because the game can't show the live multiplier. Most finishers also unlock a council agenda
# (common/council_agendas/lfg_council_agendas.txt).
#
# Supertensile modifiers only apply when Gigastructural's job-based build mode is active (global flag giga_buildcap_j);
# otherwise the matching `NOT = { has_global_flag = giga_buildcap_j }` block gives a plain megastructure fallback.

#####################################################################
# MEGAFORGE DOCTRINE - hook: grows with the number of megastructures you own; adopting unlocks agenda "The Great Project"
#####################################################################

tr_lfg_megaforge_adopt = {
	unlocks_agenda = agenda_lfg_great_project
	possible = {
		OR = {
			has_technology = giga_tech_amb_supertensiles
			has_technology = tech_mega_engineering
		}
	}
	modifier = {
		megastructure_build_speed_mult = 0.10
	}
	triggered_modifier = {
		potential = { has_global_flag = giga_buildcap_j }
		country_giga_sr_amb_megaconstruction_produces_mult = 0.15
	}
	triggered_modifier = {
		potential = { NOT = { has_global_flag = giga_buildcap_j } }
		megastructures_cost_mult = -0.05
	}
}

# country_megastructure_build_cap_add does double duty: with supertensiles off it's the vanilla
# "megastructures under construction at once" limit; with supertensiles on, Giga copies it into the
# Megaconstruction Logistics Hub limit (giga_amb_cap, giga_alternate_mega_build_effects.txt).
tr_lfg_megaforge_finish = {
	modifier = {
		ascension_perks_add = 1
		country_megastructure_build_cap_add = 1
	}
}

tr_lfg_megaforge_guilds = {
	triggered_modifier = {
		potential = { has_global_flag = giga_buildcap_j }
		planet_giga_megaengineers_produces_mult = 0.15
		planet_giga_megaengineers_upkeep_mult = 0.15
	}
	triggered_modifier = {
		potential = { NOT = { has_global_flag = giga_buildcap_j } }
		megastructure_build_speed_mult = 0.15
	}
	ai_weight = { factor = 1000 }
}

tr_lfg_megaforge_lattices = {
	triggered_modifier = {
		potential = { has_global_flag = giga_buildcap_j }
		megastructures_giga_sr_amb_megaconstruction_upkeep_mult = -0.20
		country_resource_max_giga_sr_amb_megaconstruction_add = 10000
	}
	triggered_modifier = {
		potential = { NOT = { has_global_flag = giga_buildcap_j } }
		megastructures_upkeep_mult = -0.15
		country_resource_max_add = 100000
	}
	ai_weight = { factor = 1000 }
}

# Infrastructure (buildings and districts) costs 30% more but builds 20% faster: planet_building_build_speed_mult is
# vanilla's "Infrastructure Build Speed", the same modifier Prosperity's Public Works uses.
tr_lfg_megaforge_priority = {
	modifier = {
		megastructure_build_speed_mult = 0.20
		planet_structures_cost_mult = 0.30
		planet_building_build_speed_mult = 0.20
	}
	ai_weight = { factor = 1000 }
}

# Per-job supertensile output (4.x jobs are per 100 pops). Giga's job categories also cover the drone versions
# (Supertensiles Drone, Megaconstruction Overseer). The Overlord Supertensiles Engineer is a holding job whose
# category generates no modifiers, so it can't be targeted on its own.
tr_lfg_megaforge_foundries = {
	possible = {
		has_tradition = tr_lfg_megaforge_guilds
		has_tradition = tr_lfg_megaforge_lattices
	}
	modifier = {
		country_minerals_produces_mult = -0.10
	}
	# Extra output costs alloys at each job's own rate: Technician 4 alloys : 1 supertensile, Director 10 : 5.
	triggered_modifier = {
		potential = { has_global_flag = giga_buildcap_j }
		planet_giga_megaengineers_job_giga_sr_amb_megaconstruction_produces_add = 0.25
		planet_giga_megaengineers_job_alloys_upkeep_add = 1
		planet_giga_megaengineering_overseers_giga_sr_amb_megaconstruction_produces_add = 1.5
		planet_giga_megaengineering_overseers_alloys_upkeep_add = 3
	}
	triggered_modifier = {
		potential = { NOT = { has_global_flag = giga_buildcap_j } }
		megastructures_cost_mult = -0.10
		country_alloys_produces_mult = -0.05
	}
	ai_weight = { factor = 1000 }
}

# Hook: scales with every megastructure we own (value:lfg_num_megastructures, no cap). Adopting this unlocks the
# lfg_monument_economy policy (common/policies), whose flag picks the direction:
#   Balanced (default): nothing
#   Monuments First: +1% megastructure output, -0.5% job output per megastructure
#   People First:             +0.5% job output, -1% megastructure output per megastructure
# Plus +10 supertensiles/month (Giga names the modifier "Monthly Supertensile Materials").
tr_lfg_megaforge_monuments = {
	possible = {
		has_tradition = tr_lfg_megaforge_priority
	}
	custom_tooltip = tr_lfg_megaforge_monuments_tt
	modifier = {
		country_base_giga_sr_amb_megaconstruction_produces_add = 10
	}
	triggered_modifier = {
		potential = { has_policy_flag = lfg_monuments_first }
		megastructures_produces_mult = 0.01
		planet_jobs_produces_mult = -0.005
		mult = value:lfg_num_megastructures
	}
	triggered_modifier = {
		potential = { has_policy_flag = lfg_people_first }
		planet_jobs_produces_mult = 0.005
		megastructures_produces_mult = -0.01
		mult = value:lfg_num_megastructures
	}

	ai_weight = { factor = 1000 }
}
