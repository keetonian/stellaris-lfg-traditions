# LFG Traditions - district specializations (zones).

# Hard Labour - Penal Supertensile Works: a city-slot specialization that only appears in Prison Districts on the
# penal colony (penal colonies have no City Districts), with supertensiles on and the Hard Labour finisher
# ("Penal Industries"). Prison Industry gives +50% of its jobs, like the vanilla specializations (zz_lfg_zone_overrides).
# Jobs: Penal Supertensile Laborers (common/pop_jobs/lfg_jobs.txt). Modelled on vanilla zone_foundry.
zone_lfg_penal_supertensiles = {
	icon = GFX_district_specialization_industrial
	base_buildtime = @zone_buildtime

	potential = { # planet scope
		has_modifier = penal_colony
		has_global_flag = giga_buildcap_j
		exists = owner
		owner = { has_tradition = tr_lfg_penal_finish }
	}

	unlock = { # planet scope
		has_modifier = penal_colony
	}

	resources = {
		category = planet_zones
		cost = {
			minerals = @zone_cost
		}
	}

	zone_sets = {
		urban
	}

	included_building_sets = {
		urban
		origin
	}

	triggered_district_planet_modifier = {
		potential = {
			exists = owner
		}
		job_lfg_penal_supertensile_laborer_add = @scaling_district_1_job
	}
	triggered_district_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_tradition = tr_lfg_penal_prison_industry }
		}
		job_lfg_penal_supertensile_laborer_add = @scaling_district_1_job
		mult = 0.5
	}

	planet_modifier = {
		zone_building_slots_add = 3
	}

	inline_script = {
		script = zones/shared_city_non_urban_zone_modifiers
	}

	ai_priority = 5
}
