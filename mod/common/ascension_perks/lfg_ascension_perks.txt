# LFG Traditions - ascension perks.

# Curated Progress: unlocks the Shelve edicts (common/edicts/lfg_shelve_edicts.txt), which stop specific
# repeatable techs from being offered as research options. Requires Gigastructural Constructs.
ap_lfg_curated_progress = {
	on_enabled = {
		custom_tooltip = ap_lfg_curated_progress_tt
	}
	possible = {
		custom_tooltip = { fail_text = "requires_ap_gigastructural_constructs" has_gigastructural_constructs = yes }
	}
	potential = {
		NOT = { has_ascension_perk = ap_lfg_curated_progress }
		NOT = { has_global_flag = giga_gigastructural_perk_disabled }
	}
	ai_weight = {
		factor = 0
	}
}

# Massed Assault: wider ground battles (more armies fight at once). Width is 5 + 0.2 x planet size (defines
# PLANET_COMBAT_WIDTH_*); vanilla only grants planet_combat_width_add through general traits (+1, +2, Paragon +10).
# Every general also gets leader_trait_lfg_massed_assault (army bonuses while a general leads). Needs 4 other perks first.
ap_lfg_massed_assault = {
	on_enabled = {
		custom_tooltip = ap_lfg_massed_assault_tt
		hidden_effect = { country_event = { id = lfg_massed_assault.1 } }
	}
	modifier = {
		planet_combat_width_add = @lfg_massed_assault_width
	}
	possible = {
		custom_tooltip = {
			fail_text = "lfg_requires_ascension_perks_4"	# vanilla only has _1.._3
			num_ascension_perks > 3
		}
	}
	potential = {
		NOT = { has_ascension_perk = ap_lfg_massed_assault }
	}
	ai_weight = {
		factor = 0
	}
}

# Nanite Armada: like Gigastructural's A Celestial Armada (ap_celestial_printing, same requirements plus the Nanotech
# traditions adopted, for nanite income), but nanite-made.
# Adds three engineering research options (common/technology/lfg_technology.txt): the Swarmforge megastructure
# (common/megastructures/lfg_megastructures.txt) and the nanite Attack Moon / Planetcraft hulls
# (common/ship_sizes/lfg_ship_sizes.txt), built only at a Swarmforge and paid in nanites (nanite storage has no cap).
ap_lfg_nanite_armada = {
	on_enabled = {
		custom_tooltip = ap_lfg_nanite_armada_tt
		hidden_effect = {
			if = { limit = { NOT = { has_technology = tech_lfg_swarmforge } }			add_research_option = tech_lfg_swarmforge }
			if = { limit = { NOT = { has_technology = tech_lfg_nanite_war_moon } }		add_research_option = tech_lfg_nanite_war_moon }
			if = { limit = { NOT = { has_technology = tech_lfg_nanite_planetcraft } }	add_research_option = tech_lfg_nanite_planetcraft }
		}
	}
	possible = {
		custom_tooltip = { fail_text = "requires_ap_gigastructural_constructs"	has_gigastructural_constructs = yes }
		custom_tooltip = { fail_text = "requires_ascension_perks_3"				num_ascension_perks > 2 }
		custom_tooltip = { fail_text = "requires_war_planets"					has_technology = giga_tech_war_planet }
		custom_tooltip = { fail_text = "requires_tetraengineering"				has_technology = giga_tech_tetradimensional_engineering }
		custom_tooltip = { fail_text = "lfg_requires_nanotech_adopted"			has_tradition = tr_nanotech_adopt }	# nanite income
	}
	potential = {
		NOT = { has_ascension_perk = ap_lfg_nanite_armada }
		is_nomadic = no
	}
	ai_weight = {
		factor = 0
	}
}

# Mothership: needs both celestial perks (our Nanite Armada and Gigastructural's A Celestial Armada) and the Nanotech
# traditions finished (its escort's Swarmers / Interdictors need the 4th / 5th). Adds two engineering
# research options (common/technology/lfg_technology.txt): the Progenitor Forge megastructure (around an O star) and the
# Mothership hull (common/ship_sizes/lfg_ship_sizes.txt), built only there and carrying the Nanite Swarm hangar
# (common/component_templates/lfg_components.txt).
ap_lfg_mothership = {
	on_enabled = {
		custom_tooltip = ap_lfg_mothership_tt
		hidden_effect = {
			if = { limit = { NOT = { has_technology = tech_lfg_progenitor_forge } }	add_research_option = tech_lfg_progenitor_forge }
			if = { limit = { NOT = { has_technology = tech_lfg_mothership } }			add_research_option = tech_lfg_mothership }
			if = { limit = { NOT = { has_technology = tech_lfg_cognition_core_2 } }	add_research_option = tech_lfg_cognition_core_2 }
		}
	}
	possible = {
		custom_tooltip = { fail_text = "lfg_requires_ap_nanite_armada"		has_ascension_perk = ap_lfg_nanite_armada }
		custom_tooltip = { fail_text = "lfg_requires_ap_celestial_armada"	has_ascension_perk = ap_celestial_printing }
		custom_tooltip = { fail_text = "lfg_requires_nanotech_finished"		has_tradition = tr_nanotech_finish }	# full escort
	}
	potential = {
		NOT = { has_ascension_perk = ap_lfg_mothership }
		is_nomadic = no
	}
	ai_weight = {
		factor = 0
	}
}

# Storm Sovereignty: after Galactic Wonders and Galactic Weather Control, any empire (Cosmic Storms DLC). Adds the Tempest Lodestar tech
# (common/technology/lfg_technology.txt): a megastructure that summons nexus storms (scripted action lfg_summon_nexus_storm) and
# gives its owner economy and ship bonuses inside nexus storms (generated storm_types override). Storm Chasers who take it
# become immune to cosmic storms' penalties (lfg_storm_immune_country; storm_types override, lfg_storm_immunity, lfg_storms.1).
ap_lfg_storm_sovereignty = {
	on_enabled = {
		custom_tooltip = ap_lfg_storm_sovereignty_tt
		hidden_effect = {
			if = { limit = { NOT = { has_technology = tech_lfg_tempest_lodestar } }	add_research_option = tech_lfg_tempest_lodestar }
			if = {
				limit = { has_origin = origin_storm_chasers }
				add_modifier = { modifier = lfg_storm_immunity days = -1 }
			}
		}
	}
	possible = {
		custom_tooltip = { fail_text = "lfg_requires_ap_galactic_wonders"	has_ascension_perk = ap_galactic_wonders }
		custom_tooltip = { fail_text = "lfg_requires_ap_weather_control"	has_ascension_perk = ap_weather_control }
	}
	potential = {
		NOT = { has_ascension_perk = ap_lfg_storm_sovereignty }
		has_cosmic_storms_dlc = yes
	}
	ai_weight = {
		factor = 0
	}
}

# Legendary Dragons: Here Be Dragons only, once the Sky Dragon has entrusted its young to us (vanilla's dragon_breeders
# flag, aquatics.120). Unlocks decision_lfg_legendary_dragon: anoint one Elder Dragon (lfg_drake_11) as the Legendary

		}
	}
	possible = {
		custom_tooltip = {
			fail_text = "lfg_requires_dragon_breeders"

		}
	}
	potential = {
		has_origin = origin_here_be_dragons
		NOT = { has_ascension_perk = ap_lfg_legendary_dragons }
	}
	ai_weight = {
		factor = 0
	}
}

# Shadow Transactions: Swallowed Whole (origin_lfg_sealed_ring) with Criminal Heritage, which can't take Universal
# Transactions. No other perks needed first. Unlocks the Shadow Branch Office and Remove Branch Office espionage operations
# (common/espionage_operation_types/lfg_operations.txt: from beyond the rim, opening branch offices normally always
# costs the game's 1,000-influence cap); branch offices cost half (the influence - capped anyway - and their
# buildings), +10 base intel on every empire, +1 decryption.
ap_lfg_shadow_transactions = {
	on_enabled = {
		custom_tooltip = ap_lfg_shadow_transactions_tt	# unlocks lfg_operation_shadow_branch / _remove_branch
	}
	modifier = {
		country_branch_office_influence_cost_mult = -0.5
		planet_branch_offices_cost_mult = -0.5
		add_base_country_intel = 10
		intel_decryption_add = 1
	}
	potential = {
		NOT = { has_ascension_perk = ap_lfg_shadow_transactions }
		has_origin = origin_lfg_sealed_ring
		has_valid_civic = civic_criminal_heritage
	}
	ai_weight = {
		factor = 20
	}
}
