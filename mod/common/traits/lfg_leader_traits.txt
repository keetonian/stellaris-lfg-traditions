# LFG Traditions - leader traits.

# Massed Assault (ap_lfg_massed_assault): every general of an empire with the perk gets this trait
# (events/lfg_massed_assault_events.txt), so its armies fight better whenever a general leads them. Same structure as
# vanilla's special general traits (e.g. leader_trait_cyborg, the Paragon traits): bonuses go in triggered_army_modifier.
leader_trait_lfg_massed_assault = {
	inline_script = {
		script = trait/icon
		CLASS = commander
		ICON = "GFX_leader_trait_army_veteran"
		RARITY = common
		COUNCIL = no
		TIER = none
	}
	triggered_army_modifier = {
		potential = { leader_class = commander }
		army_damage_mult = @lfg_massed_assault_general_damage
		army_health = @lfg_massed_assault_general_health
		army_morale = @lfg_massed_assault_general_morale
	}
	leader_class = { commander }
	initial = no
	randomized = no
}
