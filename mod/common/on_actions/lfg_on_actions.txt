# LFG Traditions - on_actions (merged with vanilla's lists).

on_game_start = {
	events = {
		lfg_lone_tide.1	# 75%: spawn the Lone Tide system beyond the rim
		lfg_emberfall.1	# spawn the Emberfall system somewhere in the galaxy
	}
}

on_monthly_pulse_country = {
	events = {
		lfg_resorts.1	# Galactic Destination: resort world modifier + tourist count
		lfg_shellworlders.1	# Shellworlders: artificial world modifiers
		lfg_penal.1	# Penitence: penal colony modifiers + inmate count
	}
}

# this = ship, from = planet
anomaly_success = {
	events = {
		lfg_xenoarchaeology.1	# Xenoarchaeology: minor artifacts + anomaly count
	}
}

# scope: country, from: the new leader
on_leader_spawned = {
	events = {
		lfg_massed_assault.1	# Massed Assault: give new generals the doctrine trait
	}
}

on_colony_transfer = {
	events = {
		lfg_planet.1	# strip the old owner's per-planet tradition modifiers
	}
}

on_colony_ownerless = {
	events = {
		lfg_planet.1	# strip the old owner's per-planet tradition modifiers
	}
}
