# LFG Traditions - on_actions (merged with vanilla's lists).

on_game_start = {
	events = {
		lfg_lone_tide.1	# 75%: spawn the Lone Tide system beyond the rim
		lfg_emberfall.1	# spawn the Emberfall system somewhere in the galaxy
	}
}

on_game_start_country = {
	events = {
		lfg_nanite_forge.1	# Nanite Forge origin: broken nanite harvester + home deposits
	}
}

# scope: country
on_tradition_picked = {
	events = {
		lfg_nanite_forge.3	# Nanite Forge origin: finishing Nanotech repairs the broken harvester
	}
}

on_monthly_pulse_country = {
	events = {
		lfg_resorts.1	# Galactic Destination: resort world modifier + tourist count
		lfg_shellworlders.1	# Shellworlders: artificial world modifiers
		lfg_penal.1	# Penitence: penal colony modifiers + inmate count
		lfg_nanite_forge.5	# Nanite Cognition Core: specialist automation modifier
		lfg_lineage.1	# Every Shore: monthly planet-type count
		lfg_stationcraft.1	# Stationcraft: monthly starbase / kilostructure / megastructure counts
		lfg_mothership.1	# Mothership: monthly nanite swarmers / interdictors
		lfg_storms.1	# Storm Sovereignty: immune Storm Chasers shed storm aftermath penalties
		lfg_storms.5	# Storm Sovereignty: Storm Chasers' summoned nexus storms hit other empires' fleets
		lfg_storms.7	# Storm Sovereignty: the Lodestar's system counts as inside a nexus storm
		lfg_storms.8	# Tempest tree: storms weathered / storm buildings counts
		lfg_dragons.1	# Here Be Dragons: drakes grow, and eat pops when unfed
		lfg_payback.1	# Payback: market tithes to the market host
	}
}

# this = ship, from = planet
# This = owner of fleet 1, FromFrom = fleet 1, FromFromFrom = fleet 2
on_entering_battle = {
	events = {
		lfg_mothership.3	# Mothership: deploy its nanite swarm
	}
}

anomaly_success = {
	events = {
		lfg_xenoarchaeology.1	# Xenoarchaeology: minor artifacts + anomaly count
	}
}

# scope: country, from: the new leader
on_leader_spawned = {
	events = {
		lfg_massed_assault.1	# Massed Assault: give new generals the doctrine trait
	}
}

on_colony_transfer = {
	events = {
		lfg_planet.1	# strip the old owner's per-planet tradition modifiers
	}
}

on_colony_ownerless = {
	events = {
		lfg_planet.1	# strip the old owner's per-planet tradition modifiers
	}
}

# Root = Ship
on_ship_built = {
	events = {
		lfg_mothership.6	# Mothership: one per empire (refund the extra)
		lfg_mothership.13	# Mothership: built (popup)
	}
}

	}
