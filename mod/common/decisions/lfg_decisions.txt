# LFG Traditions - planet decisions.

# Tourism - Wonders of the Galaxy: expand the resort world (an empire can only have one).
# Each use adds +2 max districts on the planet (lfg_resort_expansion, stacked via a planet variable), up to
# @lfg_resort_expansions_max times. Cost: (500 alloys + 500 minerals + 1000 unity) x (expansions made + 1), i.e.
# +500 / +500 / +1000 per previous expansion (value:lfg_resort_expansion_cost_mult), plus a flat 100 influence.
decision_lfg_expand_resort = {
	owned_planets_only = yes
	enactment_time = 360
	icon = decision_resort_colony
	resources = {
		category = decisions
		cost = {
			alloys = 500
			minerals = 500
			unity = 1000
			multiplier = value:lfg_resort_expansion_cost_mult
		}
		cost = {
			influence = 100
		}
	}
	potential = {
		has_modifier = resort_colony
		owner = { has_tradition = tr_lfg_tourism_wonders_of_the_galaxy }
	}
	allow = {
		custom_tooltip = {
			fail_text = lfg_decision_expand_resort_maxed
			NOT = { check_variable = { which = lfg_resort_expansions value >= @lfg_resort_expansions_max } }
		}
	}
	effect = {
		custom_tooltip = lfg_decision_expand_resort_effect_tt
		hidden_effect = {
			if = {
				limit = { NOT = { is_variable_set = lfg_resort_expansions } }
				set_variable = { which = lfg_resort_expansions value = 0 }
			}
			change_variable = { which = lfg_resort_expansions value = 1 }
			remove_modifier = lfg_resort_expansion
			add_modifier = {
				modifier = lfg_resort_expansion
				multiplier = lfg_resort_expansions
			}
		}
	}
	ai_weight = {
		weight = 10
	}
}

# Nanotech: the Nanite Pulse, from a star whose system has a nanite harvester (vanilla's, or the Nanite Forge origin's
# broken / ancient one) on its starbase. Costs @lfg_nanite_pulse_cost_per_body nanites per celestial body in the system
# (value:lfg_nanite_pulse_bodies); then once per @lfg_nanite_pulse_cooldown_days days for that star (timed star flag).
# Then a coin for every body in the system: heads, a world that can grow (a mining station on a harvested deposit) grows
# one harvester step (lfg_forge_harvester_step, also for vanilla's harvester); tails, or no growth,
# @lfg_nanite_pulse_payout_mult x that body's monthly harvester nanites (so early pulses cost more than they return). lfg_nanite_pulse.1 reports it all: worlds grown, nanites, cost.
decision_lfg_nanite_pulse = {
	owned_planets_only = no
	enactment_time = 0
	icon = decision_lfg_nanite_pulse
	resources = {
		category = decisions
		cost = {
			nanites = @lfg_nanite_pulse_cost_per_body
			multiplier = value:lfg_nanite_pulse_bodies
		}
	}
	potential = {
		is_star = yes
		exists = space_owner
		space_owner = {
			is_same_value = from
			has_tradition = tr_nanotech_adopt
		}
	}
	allow = {
		custom_tooltip = {
			fail_text = lfg_nanite_pulse_needs_harvester
			solar_system = {
				exists = starbase
				starbase = {
					OR = {
						has_starbase_building = nanite_harvester
						has_starbase_building = lfg_broken_nanite_harvester
						has_starbase_building = lfg_ancient_nanite_harvester
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = lfg_nanite_pulse_cooling_down
			NOT = { has_planet_flag = lfg_nanite_pulse_cooldown }
		}
	}
	effect = {
		custom_tooltip = lfg_nanite_pulse_effect_tt
		hidden_effect = {
			set_variable = { which = lfg_pulse_bodies value = value:lfg_nanite_pulse_bodies }
			set_timed_planet_flag = { flag = lfg_nanite_pulse_cooldown days = @lfg_nanite_pulse_cooldown_days }
			solar_system = { save_event_target_as = lfg_pulse_system }
			from = {
				save_event_target_as = lfg_pulse_country
				save_event_target_as = nanotech_country	# lfg_forge_harvester_step reads it
				set_variable = { which = lfg_pulse_cost value = prev.lfg_pulse_bodies }
				multiply_variable = { which = lfg_pulse_cost value = @lfg_nanite_pulse_cost_per_body }
				set_variable = { which = lfg_pulse_bodies value = prev.lfg_pulse_bodies }
				set_variable = { which = lfg_pulse_planets value = 0 }
				set_variable = { which = lfg_pulse_nanites value = 0 }
				# Mining station nanite multiplier (1 + the empire's station / nanite output modifiers), for the payouts.
				set_variable = { which = lfg_pulse_station_mult value = 1 }
				export_modifier_to_variable = { modifier = station_gatherers_produces_mult variable = lfg_pulse_mod }
				change_variable = { which = lfg_pulse_station_mult value = lfg_pulse_mod }
				export_modifier_to_variable = { modifier = station_gatherers_nanites_produces_mult variable = lfg_pulse_mod }
				change_variable = { which = lfg_pulse_station_mult value = lfg_pulse_mod }
				export_modifier_to_variable = { modifier = stations_produces_mult variable = lfg_pulse_mod }
				change_variable = { which = lfg_pulse_station_mult value = lfg_pulse_mod }
				export_modifier_to_variable = { modifier = stations_nanites_produces_mult variable = lfg_pulse_mod }
				change_variable = { which = lfg_pulse_station_mult value = lfg_pulse_mod }
				export_modifier_to_variable = { modifier = country_nanites_produces_mult variable = lfg_pulse_mod }
				change_variable = { which = lfg_pulse_station_mult value = lfg_pulse_mod }
				clear_variable = lfg_pulse_mod
			}
			# A coin for every body: heads, a world with a mining station on a harvested deposit grows one harvester step
			# (counted only if its step count went up); tails, or no growth, @lfg_nanite_pulse_payout_mult x the body's monthly
			# harvester nanites (value:lfg_pulse_deposit_nanites; nothing without a harvested deposit), with the empire's mining
			# station nanite modifiers (lfg_pulse_station_mult).
			solar_system = {
				every_system_planet = {
					set_planet_flag = lfg_pulse_pays
					random_list = {
						50 = {
							# Can grow: a mining station on a harvested deposit, i.e. any harvester step reached (vanilla's steps replace
							# their flag, so check them all) or the first deposit already there.
							lfg_count_nanite_steps = { VAR = lfg_pulse_before }
							if = {
								limit = {
									has_mining_station = yes
									OR = {
										has_deposit = d_nanite_harvester_deposit
										check_variable = { which = lfg_pulse_before value > 0 }
									}
								}
								lfg_forge_harvester_step = yes
								lfg_count_nanite_steps = { VAR = lfg_pulse_after }
								if = {
									limit = { check_variable = { which = lfg_pulse_after value > lfg_pulse_before } }
									remove_planet_flag = lfg_pulse_pays
									event_target:lfg_pulse_country = { change_variable = { which = lfg_pulse_planets value = 1 } }
								}
								clear_variable = lfg_pulse_after
							}
							clear_variable = lfg_pulse_before
						}
						50 = { }
					}
					if = {
						limit = { has_planet_flag = lfg_pulse_pays }
						remove_planet_flag = lfg_pulse_pays
						set_variable = { which = lfg_pulse_pay value = value:lfg_pulse_deposit_nanites }
						multiply_variable = { which = lfg_pulse_pay value = @lfg_nanite_pulse_payout_mult }
						event_target:lfg_pulse_country = {
							set_variable = { which = lfg_pulse_pay value = prev.lfg_pulse_pay }
							multiply_variable = { which = lfg_pulse_pay value = lfg_pulse_station_mult }
							change_variable = { which = lfg_pulse_nanites value = lfg_pulse_pay }
							clear_variable = lfg_pulse_pay
						}
						clear_variable = lfg_pulse_pay
					}
				}
			}
			from = {
				add_resource = {
					nanites = 1
					mult = lfg_pulse_nanites
				}
				country_event = { id = lfg_nanite_pulse.1 }
			}
		}
	}
	ai_weight = {
		weight = 0
	}
}

# Mothership crew: like Gigastructural's decision_systemcraft_crew (4,000 pops for a systemcraft), a planet with more than
# 2,000 pops gives @lfg_mothership_crew_hundreds x 100 of them (kill_single_pop, 100 each) to crew the Mothership. Sets
# the country flag its construction needs (lfg_mothership possible_construction); a finished Mothership uses it up.
decision_lfg_mothership_crew = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = decision_lfg_nanite_pulse
	potential = {
		owner = {
			has_technology = tech_lfg_mothership
			NOT = { has_country_flag = lfg_mothership_crew_ready }
		}
	}
	allow = {
		custom_tooltip = {
			fail_text = lfg_mothership_crew_needs_pops
			count_owned_pop_amount = { count >= 2100 }
		}
	}
	effect = {
		custom_tooltip = lfg_mothership_crew_effect_tt
		hidden_effect = {
			owner = { set_country_flag = lfg_mothership_crew_ready }
			while = {
				count = @lfg_mothership_crew_hundreds
				random_owned_pop_group = {
					kill_single_pop = yes
				}
			}
		}
	}
	ai_weight = {
		weight = 0
	}
}

# Legendary Dragons (ap_lfg_legendary_dragons), on the capital: anoint one of our Elder Dragons as the Legendary Dragon
# (lfg_dragons.3 picks it). Only while we have none.
decision_lfg_legendary_dragon = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = decision_lfg_legendary_dragon
	potential = {
		is_capital = yes
		owner = { has_ascension_perk = ap_lfg_legendary_dragons }
	}
	allow = {
		owner = {
			custom_tooltip = {
				fail_text = lfg_legendary_dragon_exists
				NOT = { lfg_has_legendary_dragon = yes }
			}
			custom_tooltip = {
				fail_text = lfg_needs_elder_dragon
				any_owned_ship = { lfg_is_elder_drake = yes }
			}
		}
	}
	effect = {
		custom_tooltip = decision_lfg_legendary_dragon_effect_tt
		hidden_effect = {
			owner = { country_event = { id = lfg_dragons.3 } }
		}
	}
	ai_weight = {
		weight = 1
	}
}
