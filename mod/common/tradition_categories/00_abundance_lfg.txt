# LFG Traditions - tradition tree (granary).
# All trees use the vanilla tree_21_11 layout (same as Supremacy):
#   slots 1 2 3 on top; slot 4 requires 1 + 2; slot 5 requires 3.
# The order of `traditions` maps to slots 1..5. Prerequisites live in common/traditions/lfg_traditions.txt.
# One tree per file, named 00_<English tree name>_lfg.txt, so the traditions window sorts it alphabetically
# among vanilla's regular trees (00_<tree>.txt) instead of after the ascension trees (01_*).

# The food stockpile as political power. Only for empires whose main species eats food (lfg_main_species_eats_food).
tradition_lfg_abundance = {
	desc = {
		trigger = { is_hive_empire = no }
		text = tradition_lfg_abundance_desc
	}
	desc = {
		trigger = { is_hive_empire = yes }
		text = tradition_lfg_abundance_hive_desc
	}
	potential = {
		lfg_main_species_eats_food = yes
	}
	tree_template = "tree_21_11"
	adoption_bonus = "tr_lfg_abundance_adopt"
	finish_bonus = "tr_lfg_abundance_finish"
	traditions = {
		"tr_lfg_abundance_tithe_barns"
		"tr_lfg_abundance_seed_vaults"
		"tr_lfg_abundance_bountiful_fields"
		"tr_lfg_abundance_grain_reserves"
		"tr_lfg_abundance_harvest_courts"
	}
	ai_weight = {
		factor = 4
		modifier = {
			factor = 2
			OR = {
				has_origin = origin_evolutionary_predators
				has_valid_civic = civic_agrarian_idyll
			}
		}
		modifier = {
			factor = @ap_pending_tradition_suppress
			has_ap_pending = yes
		}
	}
}
