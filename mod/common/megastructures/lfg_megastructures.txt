	}

# Mothership (ap_lfg_mothership): the Progenitor Forge, around an O star (Gigastructural's giga_is_o_star_for_megas:
# its own O stars and Real Space's). Three stages, 50,000,000 nanites in all (@lfg_progenitor_forge_*_nanites).
# Finished, it's a shipyard (like the Swarmforge, 300 ships at once) and the only place Motherships can be built; it
# also builds the nanite Attack Moons and Planetcraft. It's built around the star, like Gigastructural's Hyperstructural
# Assembly: nanite station and hive rings, Giga's assembly yards and grey goo clouds (mod/gfx/models/ships/lfg_megastructures.asset).
lfg_progenitor_forge_0 = {
	entity = "giga_consite_7_7_entity"
	construction_entity = "giga_consite_7_7_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	build_time = 1800
	resources = {
		category = megastructures
		cost = {
			alloys = 50000
			nanites = @lfg_progenitor_forge_0_nanites
			influence = 200
		}
		upkeep = {
			energy = 50
		}
	}

	prerequisites = { "tech_lfg_progenitor_forge" }

	potential = {
		has_ascension_perk = ap_lfg_mothership
	}

	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_no_existing_megastructure"
			has_no_non_gate_megastructure = yes
		}
	}

	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = lfg_requires_o_star
				giga_is_o_star_for_megas = yes
			}
		}
	}

	# The site done (this = the system, from = the builder): the swarms take the rest of the system apart for bulk matter
	# (lfg_progenitor_forge_consume_system), like Gigastructural's Hyperstructural Assembly clearing its system.
	on_build_complete = {
		from = { set_country_flag = has_built_or_repaired_megastructure }
		lfg_progenitor_forge_consume_system = yes
	}

	ai_weight = {
		weight = 0
	}
}

	entity = "lfg_tempest_lodestar_entity"
	construction_entity = "lfg_tempest_lodestar_construction_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	build_time = 1800
	upgrade_from = {
		lfg_tempest_lodestar_0
	}
	resources = {
		category = megastructures
		cost = {
			alloys = 35000
			energy = 25000
		}
		upkeep = {
			energy = 100
		}
	}
	prerequisites = { "tech_lfg_tempest_lodestar" }
	# A captured site can't be finished without the perk.
	potential = {
		has_ascension_perk = ap_lfg_storm_sovereignty
	}
	# Its effects are scripted (storm overrides, lfg_lodestar_storm_pull), so its description (lfg_tempest_lodestar_1_DESC)
	# lists them: custom_tooltip and custom_tooltip_with_modifiers both showed twice in the hover tooltip in game.
	# Summon Nexus Storm button (common/scripted_actions/lfg_scripted_actions.txt).
	scripted_action = { lfg_summon_nexus_storm }
	on_build_complete = {
		from = { set_country_flag = has_built_or_repaired_megastructure }
		# Its star pulls storms toward the system (like vanilla's storm attraction buildings: a planet modifier with
		# system_storm_influence_add plus an influence field around the system).
		fromfrom.planet = {
			set_planet_flag = lfg_tempest_lodestar
			add_modifier = { modifier = lfg_lodestar_storm_pull days = -1 }
			create_influence_field_if_needed = yes
		}
	}
}
