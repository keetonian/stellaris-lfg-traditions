# LFG Traditions - civics. Every civic here uses our own icon from art/make_civic_icons.py: a vanilla glyph recoloured
# to the mod's civic colour, so they all match.

# Biologists research less society but some physics and engineering (per Biologist job: 3 society -> 2 society
# + 0.4 physics + 0.4 engineering). Covers the Biologist job swaps too (Rangers, Xeno-Zoo Keepers, Stratovent
# Researchers, Archaeo-Engineers). Materialist, non-machine species; the materialist ethic already rules out gestalts.
civic_lfg_applied_biosciences = {
	icon = "gfx/interface/icons/governments/civics/civic_lfg_applied_biosciences.dds"
	potential = {
		ethics = { NOT = { value = ethic_gestalt_consciousness } }
	}
	possible = {
		ethics = {
			OR = {
				text = civic_tooltip_materialist
				value = ethic_materialist
				value = ethic_fanatic_materialist
			}
		}
		species_archetype = { NOT = { value = MACHINE } }
	}
	modifier = {
		planet_biologists_society_research_produces_add = -1
		planet_biologists_physics_research_produces_add = 0.4
		planet_biologists_engineering_research_produces_add = 0.4
	}
	random_weight = { base = @civic_default_random_weight }
	ai_weight = {
		base = @ai_civic_default_base_weight
		modifier = {
			factor = @ai_civic_personality_match_factor
			has_ai_personality = erudite_explorers
		}
	}
}

# Machine origin (individual or gestalt: species archetype MACHINE). Home system lfg_nanite_forge_start (B star with a
# 22-energy deposit, eight size 40 molten worlds); unlocks building_lfg_nanite_automation; a broken nanite harvester on the home starbase doubles the system's nanite deposits every 3 years up
# to step 7 until Nanotech is finished (common/starbase_buildings, events/lfg_nanite_forge_events.txt). Needs Machine Age
# (nanites and the Nanotech tree).
origin_lfg_nanite_forge = {
	is_origin = yes
	playable = { host_has_dlc = "The Machine Age" }
	ai_playable = { host_has_dlc = "The Machine Age" }
	# Art: art/make_origin_art.py (nanite diamond icon; a world engulfed by nanites).
	icon = "gfx/interface/icons/origins/lfg_origin_nanite_forge.dds"
	picture = GFX_origin_lfg_nanite_forge
	flags = { custom_start_screen }
	initializers = { lfg_nanite_forge_start }
	possible = {
		species_archetype = { value = MACHINE }
		is_nomadic = no
	}
	description = "origin_tooltip_lfg_nanite_forge_effects"
	# The founder species always has Nanite Substrate: shown added and locked in the species designer (as vanilla's
	# Evolutionary Predators forces trait_malleable_genes).
	traits = {
		trait = trait_lfg_nanite_substrate
	}
	# Orbital stations (mining and research) +30%. The pop assembly downside is on the Nanite Substrate trait.
	modifier = {
		station_gatherers_produces_mult = 0.30
		station_researchers_produces_mult = 0.30
	}
	random_weight = { base = 1 }
}
