# LFG Traditions - Nanite Forge origin: the home system's special nanite harvester.

# Planet scope, needs event_target:nanotech_country. One doubling step of the origin's harvester (lfg_nanite_forge.2).
# Vanilla's steps (cybernetics.5054: same deposits, flags, size limits and 50% odds, the failed 50% paying out) and our
# extra steps 10-12 (same as every harvester's), then two more of its own: 13 (size >= 45: 2 x 409.6 = 819.2, in our
# d_lfg_nanite_harvester_deposit_huge instead of 32 vanilla deposits), 14 (size >= 50: 1,638.4), 15 (size >= 55: 2 x
# 1,638.4 = 3,276.8, in d_lfg_nanite_harvester_deposit_vast) and 16 (size >= 60: 6,553.6).
# Broken (before Nanotech): no further than step 7. Steps 8+ need the star flag lfg_nanite_harvester_restored (set when
# it becomes the Ancient Nanite Harvester), or for steps 8-12 vanilla's harvester (the Nanite Pulse steps single worlds); planets on step 9+ that can't grow pay out like vanilla's.
# Planets keep nanites_doubled_9 from step 9 on, so vanilla's harvester upkeep value counts them.
lfg_forge_harvester_step = {
	if = {
		limit = {
			NOR = {
				has_planet_flag = nanites_doubled_1
				has_planet_flag = nanites_doubled_2
				has_planet_flag = nanites_doubled_3
				has_planet_flag = nanites_doubled_4
				has_planet_flag = nanites_doubled_5
				has_planet_flag = nanites_doubled_6
				has_planet_flag = nanites_doubled_7
				has_planet_flag = nanites_doubled_8
				has_planet_flag = nanites_doubled_9
			}
		}
		add_deposit = d_nanite_harvester_deposit
		set_planet_flag = nanites_doubled_1
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_1
			planet_size >= 3
		}
		add_deposit = d_nanite_harvester_deposit
		add_deposit = d_nanite_harvester_deposit
		set_planet_flag = nanites_doubled_2
		remove_planet_flag = nanites_doubled_1
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_2
			planet_size > 6
		}
		add_deposit = d_nanite_harvester_deposit
		add_deposit = d_nanite_harvester_deposit
		add_deposit = d_nanite_harvester_deposit
		add_deposit = d_nanite_harvester_deposit
		set_planet_flag = nanites_doubled_3
		remove_planet_flag = nanites_doubled_2
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_3
			planet_size >= 9
		}
		every_deposit = {
			limit = { is_deposit_type = d_nanite_harvester_deposit }
			remove_deposit = yes
		}
		add_deposit = d_nanite_harvester_deposit_regular
		set_planet_flag = nanites_doubled_4
		remove_planet_flag = nanites_doubled_3
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_4
			planet_size >= 12
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_nanite_harvester_deposit_regular
				set_planet_flag = nanites_doubled_5
				remove_planet_flag = nanites_doubled_4
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_5
			planet_size >= 15
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_nanite_harvester_deposit_regular
				add_deposit = d_nanite_harvester_deposit_regular
				set_planet_flag = nanites_doubled_6
				remove_planet_flag = nanites_doubled_5
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_6
			planet_size >= 18
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_nanite_harvester_deposit_regular
				add_deposit = d_nanite_harvester_deposit_regular
				add_deposit = d_nanite_harvester_deposit_regular
				add_deposit = d_nanite_harvester_deposit_regular
				set_planet_flag = nanites_doubled_7
				remove_planet_flag = nanites_doubled_6
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_7
			planet_size >= 21
			solar_system = {
				OR = {
					has_star_flag = lfg_nanite_harvester_restored
					starbase = { has_starbase_building = nanite_harvester }	# vanilla's harvester, through the Nanite Pulse
				}
			}
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				every_deposit = {
					limit = { is_deposit_type = d_nanite_harvester_deposit_regular }
					remove_deposit = yes
				}
				add_deposit = d_nanite_harvester_deposit_large
				set_planet_flag = nanites_doubled_8
				remove_planet_flag = nanites_doubled_7
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_8
			planet_size >= 25
			solar_system = {
				OR = {
					has_star_flag = lfg_nanite_harvester_restored
					starbase = { has_starbase_building = nanite_harvester }	# vanilla's harvester, through the Nanite Pulse
				}
			}
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_nanite_harvester_deposit_large
				set_planet_flag = nanites_doubled_9
				remove_planet_flag = nanites_doubled_8
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_9
			NOT = { has_planet_flag = lfg_nanites_doubled_10 }
			planet_size >= @lfg_nanite_step_10_size
			solar_system = {
				OR = {
					has_star_flag = lfg_nanite_harvester_restored
					starbase = { has_starbase_building = nanite_harvester }	# vanilla's harvester, through the Nanite Pulse
				}
			}
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_nanite_harvester_deposit_large
				add_deposit = d_nanite_harvester_deposit_large
				set_planet_flag = lfg_nanites_doubled_10
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = lfg_nanites_doubled_10
			NOT = { has_planet_flag = lfg_nanites_doubled_11 }
			planet_size >= @lfg_nanite_step_11_size
			solar_system = {
				OR = {
					has_star_flag = lfg_nanite_harvester_restored
					starbase = { has_starbase_building = nanite_harvester }	# vanilla's harvester, through the Nanite Pulse
				}
			}
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_nanite_harvester_deposit_large
				add_deposit = d_nanite_harvester_deposit_large
				add_deposit = d_nanite_harvester_deposit_large
				add_deposit = d_nanite_harvester_deposit_large
				set_planet_flag = lfg_nanites_doubled_11
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = lfg_nanites_doubled_11
			NOT = { has_planet_flag = lfg_nanites_doubled_12 }
			planet_size >= @lfg_nanite_step_12_size
			solar_system = {
				OR = {
					has_star_flag = lfg_nanite_harvester_restored
					starbase = { has_starbase_building = nanite_harvester }	# vanilla's harvester, through the Nanite Pulse
				}
			}
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				while = { count = 8 add_deposit = d_nanite_harvester_deposit_large }
				set_planet_flag = lfg_nanites_doubled_12
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = lfg_nanites_doubled_12
			NOT = { has_planet_flag = lfg_nanites_doubled_13 }
			planet_size >= @lfg_nanite_step_13_size
			solar_system = { has_star_flag = lfg_nanite_harvester_restored }
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				every_deposit = {
					limit = { is_deposit_type = d_nanite_harvester_deposit_large }
					remove_deposit = yes
				}
				add_deposit = d_lfg_nanite_harvester_deposit_huge
				add_deposit = d_lfg_nanite_harvester_deposit_huge
				set_planet_flag = lfg_nanites_doubled_13
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = lfg_nanites_doubled_13
			NOT = { has_planet_flag = lfg_nanites_doubled_14 }
			planet_size >= @lfg_nanite_step_14_size
			solar_system = { has_star_flag = lfg_nanite_harvester_restored }
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_lfg_nanite_harvester_deposit_huge
				add_deposit = d_lfg_nanite_harvester_deposit_huge
				set_planet_flag = lfg_nanites_doubled_14
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = lfg_nanites_doubled_14
			NOT = { has_planet_flag = lfg_nanites_doubled_15 }
			planet_size >= @lfg_nanite_step_15_size
			solar_system = { has_star_flag = lfg_nanite_harvester_restored }
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				every_deposit = {
					limit = { is_deposit_type = d_lfg_nanite_harvester_deposit_huge }
					remove_deposit = yes
				}
				add_deposit = d_lfg_nanite_harvester_deposit_vast
				add_deposit = d_lfg_nanite_harvester_deposit_vast
				set_planet_flag = lfg_nanites_doubled_15
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = lfg_nanites_doubled_15
			NOT = { has_planet_flag = lfg_nanites_doubled_16 }
			planet_size >= @lfg_nanite_step_16_size
			solar_system = { has_star_flag = lfg_nanite_harvester_restored }
		}
		random_list = {
			50 = { nanotech_deposit_nanite_generation = yes }
			50 = {
				add_deposit = d_lfg_nanite_harvester_deposit_vast
				add_deposit = d_lfg_nanite_harvester_deposit_vast
				set_planet_flag = lfg_nanites_doubled_16
			}
		}
	}
	else_if = {
		limit = {
			has_planet_flag = nanites_doubled_9
			solar_system = { has_star_flag = lfg_nanite_harvester_restored }
		}
		nanotech_deposit_nanite_generation = yes
	}
}

# Starbase scope. Finishing Nanotech: swap the Broken Nanite Harvester for the Ancient Nanite Harvester in the slot
# lfg_nanite_forge.6 remembered. Both are ours, so the swap only has to pass our own `possible` (country flag
# lfg_placing_forge_harvester); the timed star flag stops the old building's on_destroyed cleanup (lfg_nanite_forge.4).
lfg_restore_forge_harvester = {
	solar_system = {
		set_timed_star_flag = { flag = lfg_nanite_harvester_swapping days = 2 }
		set_star_flag = lfg_nanite_harvester_restored
	}
	owner = { set_country_flag = lfg_placing_forge_harvester }
	if = {
		limit = { solar_system = { has_star_flag = lfg_broken_harvester_slot_1 } }
		set_starbase_building = { slot = 1 building = lfg_ancient_nanite_harvester }
	}
	else_if = {
		limit = { solar_system = { has_star_flag = lfg_broken_harvester_slot_2 } }
		set_starbase_building = { slot = 2 building = lfg_ancient_nanite_harvester }
	}
	else_if = {
		limit = { solar_system = { has_star_flag = lfg_broken_harvester_slot_3 } }
		set_starbase_building = { slot = 3 building = lfg_ancient_nanite_harvester }
	}
	else_if = {
		limit = { solar_system = { has_star_flag = lfg_broken_harvester_slot_4 } }
		set_starbase_building = { slot = 4 building = lfg_ancient_nanite_harvester }
	}
	else_if = {
		limit = { solar_system = { has_star_flag = lfg_broken_harvester_slot_5 } }
		set_starbase_building = { slot = 5 building = lfg_ancient_nanite_harvester }
	}
	else_if = {
		limit = { solar_system = { has_star_flag = lfg_broken_harvester_slot_6 } }
		set_starbase_building = { slot = 6 building = lfg_ancient_nanite_harvester }
	}
	owner = { remove_country_flag = lfg_placing_forge_harvester }
}

# System scope. The harvester was removed: clear every harvester deposit and step flag in the system (like vanilla's
# cybernetics.5051, plus our huge / vast deposits and step flags 10-16).
lfg_clear_forge_harvester = {
	remove_star_flag = lfg_nanite_harvester_restored
	every_system_planet = {
		every_deposit = {
			limit = {
				OR = {
					is_deposit_type = d_nanite_harvester_deposit
					is_deposit_type = d_nanite_harvester_deposit_regular
					is_deposit_type = d_nanite_harvester_deposit_large
					is_deposit_type = d_lfg_nanite_harvester_deposit_huge
					is_deposit_type = d_lfg_nanite_harvester_deposit_vast
				}
			}
			remove_deposit = yes
		}
		remove_planet_flag = nanites_doubled_1
		remove_planet_flag = nanites_doubled_2
		remove_planet_flag = nanites_doubled_3
		remove_planet_flag = nanites_doubled_4
		remove_planet_flag = nanites_doubled_5
		remove_planet_flag = nanites_doubled_6
		remove_planet_flag = nanites_doubled_7
		remove_planet_flag = nanites_doubled_8
		remove_planet_flag = nanites_doubled_9
		remove_planet_flag = lfg_nanites_doubled_10
		remove_planet_flag = lfg_nanites_doubled_11
		remove_planet_flag = lfg_nanites_doubled_12
		remove_planet_flag = lfg_nanites_doubled_13
		remove_planet_flag = lfg_nanites_doubled_14
		remove_planet_flag = lfg_nanites_doubled_15
		remove_planet_flag = lfg_nanites_doubled_16
	}
}

# System scope, with the origin's harvester: one doubling step for each deposit planet with a mining station
# (lfg_nanite_forge.2 every 3 years; the Nanite Pulse decision).
lfg_forge_harvester_grow = {
	starbase.owner = { save_event_target_as = nanotech_country }
	every_system_planet = {
		limit = {
			has_mining_station = yes
			OR = {
				has_deposit = d_nanite_harvester_deposit
				has_planet_flag = nanites_doubled_1
				has_planet_flag = nanites_doubled_2
				has_planet_flag = nanites_doubled_3
				has_planet_flag = nanites_doubled_4
				has_planet_flag = nanites_doubled_5
				has_planet_flag = nanites_doubled_6
				has_planet_flag = nanites_doubled_7
				has_planet_flag = nanites_doubled_8
				has_planet_flag = nanites_doubled_9
			}
		}
		lfg_forge_harvester_step = yes
	}
}

# Planet scope: the highest harvester step this world has reached, into planet variable $VAR$ (vanilla's steps replace
# their flag, nanites_doubled_1 -> _2 ...; our steps 10-16 add theirs while it keeps nanites_doubled_9).
lfg_count_nanite_steps = {
	set_variable = { which = $VAR$ value = 0 }
	if = { limit = { has_planet_flag = nanites_doubled_1 } set_variable = { which = $VAR$ value = 1 } }
	if = { limit = { has_planet_flag = nanites_doubled_2 } set_variable = { which = $VAR$ value = 2 } }
	if = { limit = { has_planet_flag = nanites_doubled_3 } set_variable = { which = $VAR$ value = 3 } }
	if = { limit = { has_planet_flag = nanites_doubled_4 } set_variable = { which = $VAR$ value = 4 } }
	if = { limit = { has_planet_flag = nanites_doubled_5 } set_variable = { which = $VAR$ value = 5 } }
	if = { limit = { has_planet_flag = nanites_doubled_6 } set_variable = { which = $VAR$ value = 6 } }
	if = { limit = { has_planet_flag = nanites_doubled_7 } set_variable = { which = $VAR$ value = 7 } }
	if = { limit = { has_planet_flag = nanites_doubled_8 } set_variable = { which = $VAR$ value = 8 } }
	if = { limit = { has_planet_flag = nanites_doubled_9 } set_variable = { which = $VAR$ value = 9 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_10 } set_variable = { which = $VAR$ value = 10 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_11 } set_variable = { which = $VAR$ value = 11 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_12 } set_variable = { which = $VAR$ value = 12 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_13 } set_variable = { which = $VAR$ value = 13 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_14 } set_variable = { which = $VAR$ value = 14 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_15 } set_variable = { which = $VAR$ value = 15 } }
	if = { limit = { has_planet_flag = lfg_nanites_doubled_16 } set_variable = { which = $VAR$ value = 16 } }
}
