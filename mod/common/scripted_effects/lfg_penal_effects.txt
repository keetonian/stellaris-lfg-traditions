# LFG Traditions - Penitence effects.

# Planet scope. Keeps static modifier $MODIFIER$ on this planet exactly while it is the owner's penal colony and the
# owner has $TRADITION$.
lfg_sync_penal_modifier = {
	if = {
		limit = {
			has_modifier = penal_colony
			owner = { has_tradition = $TRADITION$ }
		}
		if = {
			limit = { NOT = { has_modifier = $MODIFIER$ } }
			add_modifier = { modifier = $MODIFIER$ days = -1 }
		}
	}
	else_if = {
		limit = { has_modifier = $MODIFIER$ }
		remove_modifier = $MODIFIER$
	}
}

# "Unemployed" below means Civilians: in 4.x a regular empire's jobless pops take the uncapped Civilian job, so
# num_unemployed is always 0 for them; count num_assigned_jobs = { job = civilian } instead (as Galactic Destination
# does for Tourists).

# Planet scope. Idle Hands: lfg_penal_idle_hands with multiplier = whole hundreds of unemployed pops here
# (planet variable lfg_unemployed_hundreds). Re-added only when that count changes.
lfg_sync_penal_unemployed = {
	set_variable = { which = lfg_unemployed_hundreds_now value = 0 }
	if = {
		limit = {
			has_modifier = penal_colony
			owner = { has_tradition = tr_lfg_penitence_idle_hands }
		}
		export_trigger_value_to_variable = {
			trigger = num_assigned_jobs
			parameters = {
				job = civilian
			}
			variable = lfg_unemployed_hundreds_now
		}
		divide_variable = { which = lfg_unemployed_hundreds_now value = 100 }
		floor_variable = lfg_unemployed_hundreds_now
		lfg_penal_idle_hands_attrition = yes
	}
	if = {
		limit = {
			OR = {
				NOT = { check_variable = { which = lfg_unemployed_hundreds value = lfg_unemployed_hundreds_now } }
				AND = {
					check_variable = { which = lfg_unemployed_hundreds_now value > 0 }
					NOT = { has_modifier = lfg_penal_idle_hands }
				}
			}
		}
		remove_modifier = lfg_penal_idle_hands
		set_variable = { which = lfg_unemployed_hundreds value = lfg_unemployed_hundreds_now }
		if = {
			limit = { check_variable = { which = lfg_unemployed_hundreds value > 0 } }
			add_modifier = { modifier = lfg_penal_idle_hands days = -1 multiplier = lfg_unemployed_hundreds }
		}
	}	# Fewer than 100 unemployed while holding Idle Hands: @lfg_idle_hands_empty_penalty job output instead.
	if = {
		limit = {
			has_modifier = penal_colony
			owner = { has_tradition = tr_lfg_penitence_idle_hands }
			check_variable = { which = lfg_unemployed_hundreds_now value < 1 }
		}
		if = {
			limit = { NOT = { has_modifier = lfg_penal_no_idle_hands } }
			add_modifier = { modifier = lfg_penal_no_idle_hands days = -1 }
		}
	}
	else_if = {
		limit = { has_modifier = lfg_penal_no_idle_hands }
		remove_modifier = lfg_penal_no_idle_hands
	}
}

# Planet scope (the penal colony, owner has Idle Hands). Attrition: @lfg_idle_hands_attrition of the unemployed die
# each month. Pop groups mix employed and unemployed pops, so every group loses the same share of its size, adding up
# to that many pops (the jobs they leave are refilled by the unemployed): share = unemployed * attrition / population.
lfg_penal_idle_hands_attrition = {
	export_trigger_value_to_variable = {
		trigger = num_assigned_jobs
		parameters = {
			job = civilian
		}
		variable = lfg_attrition_share
	}
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = { always = yes }
		}
		variable = lfg_attrition_population
	}
	if = {
		limit = {
			check_variable = { which = lfg_attrition_share value > 0 }
			check_variable = { which = lfg_attrition_population value > 0 }
		}
		multiply_variable = { which = lfg_attrition_share value = @lfg_idle_hands_attrition }
		divide_variable = { which = lfg_attrition_share value = lfg_attrition_population }
		every_owned_pop_group = {
			export_trigger_value_to_variable = {
				trigger = pop_group_size
				variable = lfg_attrition_kill
			}
			multiply_variable = { which = lfg_attrition_kill value = prev.lfg_attrition_share }
			floor_variable = lfg_attrition_kill
			if = {
				limit = { check_variable = { which = lfg_attrition_kill value > 0 } }
				kill_pop_group = {
					pop_group = this
					amount = lfg_attrition_kill
				}
			}
		}
	}
}

# Planet scope (a planet that isn't the penal colony; event_target:lfg_penal_target = the penal colony). Transportation:
# ship @lfg_transportation_share of this planet's unemployed (at most @lfg_transportation_max) to the penal colony,
# whether or not it has jobs or housing. Pop groups mix employed and unemployed pops, so every group sends the same
# share of its size (the jobs they leave are refilled by the unemployed), adding up to that many pops:
# share = min(unemployed * 10%, 300) / population. Each group moves into a new group on the penal colony based on it
# (create_pop_group + transfer_pop_amount, as in lfg_split_species).
lfg_penal_ship_unemployed = {
	export_trigger_value_to_variable = {
		trigger = num_assigned_jobs
		parameters = {
			job = civilian
		}
		variable = lfg_ship_share
	}
	multiply_variable = { which = lfg_ship_share value = @lfg_transportation_share }
	floor_variable = lfg_ship_share
	if = {
		limit = { check_variable = { which = lfg_ship_share value > @lfg_transportation_max } }
		set_variable = { which = lfg_ship_share value = @lfg_transportation_max }
	}
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = { always = yes }
		}
		variable = lfg_ship_population
	}
	if = {
		limit = {
			check_variable = { which = lfg_ship_share value > 0 }
			check_variable = { which = lfg_ship_population value > 0 }
		}
		divide_variable = { which = lfg_ship_share value = lfg_ship_population }
		every_owned_pop_group = {
			export_trigger_value_to_variable = {
				trigger = pop_group_size
				variable = lfg_ship_amount
			}
			multiply_variable = { which = lfg_ship_amount value = prev.lfg_ship_share }
			floor_variable = lfg_ship_amount
			if = {
				limit = { check_variable = { which = lfg_ship_amount value > 0 } }
				save_event_target_as = lfg_ship_source
				event_target:lfg_penal_target = {
					create_pop_group = {
						pop_group = event_target:lfg_ship_source
						size = 0
						effect = {
							set_variable = { which = lfg_ship_amount value = event_target:lfg_ship_source.lfg_ship_amount }
							transfer_pop_amount = {
								source = event_target:lfg_ship_source
								target = this
								amount = lfg_ship_amount
							}
						}
					}
				}
			}
		}
	}
}
