# LFG Traditions - Hard Labour effects.

# Planet scope. Keeps static modifier $MODIFIER$ on this planet exactly while it is the owner's penal colony and the
# owner has $TRADITION$.
lfg_sync_penal_modifier = {
	if = {
		limit = {
			has_modifier = penal_colony
			owner = { has_tradition = $TRADITION$ }
		}
		if = {
			limit = { NOT = { has_modifier = $MODIFIER$ } }
			add_modifier = { modifier = $MODIFIER$ days = -1 }
		}
	}
	else_if = {
		limit = { has_modifier = $MODIFIER$ }
		remove_modifier = $MODIFIER$
	}
}

# Planet scope. Penal Industries (Hard Labour finisher): lfg_penal_prison_industry (or _crafter) with multiplier = Prison Districts here
# (planet variable lfg_prison_districts). Re-added only when the count changes, like lfg_sync_arc_neighbours.
lfg_sync_prison_industry = {
	set_variable = { which = lfg_prison_districts_now value = 0 }
	if = {
		limit = {
			has_modifier = penal_colony
			owner = { has_tradition = tr_lfg_penal_finish }
		}
		export_trigger_value_to_variable = {
			trigger = num_districts
			parameters = {
				type = district_prison
			}
			variable = lfg_prison_districts_now
		}
	}
	if = {
		limit = {
			OR = {
				NOT = { check_variable = { which = lfg_prison_districts value = lfg_prison_districts_now } }
				AND = {
					check_variable = { which = lfg_prison_districts_now value > 0 }
					NOR = {
						has_modifier = lfg_penal_prison_industry
						has_modifier = lfg_penal_prison_industry_crafter
					}
				}
			}
		}
		remove_modifier = lfg_penal_prison_industry
		remove_modifier = lfg_penal_prison_industry_crafter
		set_variable = { which = lfg_prison_districts value = lfg_prison_districts_now }
		if = {
			limit = { check_variable = { which = lfg_prison_districts value > 0 } }
			if = {
				limit = { owner = { is_crafter_empire = yes } }
				add_modifier = { modifier = lfg_penal_prison_industry_crafter days = -1 multiplier = lfg_prison_districts }
			}
			else = {
				add_modifier = { modifier = lfg_penal_prison_industry days = -1 multiplier = lfg_prison_districts }
			}
		}
	}
}

# Planet scope. Idle Hands: lfg_penal_idle_hands with multiplier = whole hundreds of unemployed pops here
# (planet variable lfg_unemployed_hundreds). Re-added only when that count changes.
lfg_sync_penal_unemployed = {
	set_variable = { which = lfg_unemployed_hundreds_now value = 0 }
	if = {
		limit = {
			has_modifier = penal_colony
			owner = { has_tradition = tr_lfg_penal_idle_hands }
		}
		export_trigger_value_to_variable = {
			trigger = num_unemployed
			variable = lfg_unemployed_hundreds_now
		}
		divide_variable = { which = lfg_unemployed_hundreds_now value = 100 }
		floor_variable = lfg_unemployed_hundreds_now
		lfg_penal_idle_hands_attrition = yes
	}
	if = {
		limit = {
			OR = {
				NOT = { check_variable = { which = lfg_unemployed_hundreds value = lfg_unemployed_hundreds_now } }
				AND = {
					check_variable = { which = lfg_unemployed_hundreds_now value > 0 }
					NOT = { has_modifier = lfg_penal_idle_hands }
				}
			}
		}
		remove_modifier = lfg_penal_idle_hands
		set_variable = { which = lfg_unemployed_hundreds value = lfg_unemployed_hundreds_now }
		if = {
			limit = { check_variable = { which = lfg_unemployed_hundreds value > 0 } }
			add_modifier = { modifier = lfg_penal_idle_hands days = -1 multiplier = lfg_unemployed_hundreds }
		}
	}
}

# Planet scope (the penal colony, owner has Idle Hands). Attrition: @lfg_idle_hands_attrition of the unemployed die
# each month. Pop groups mix employed and unemployed pops, so every group loses the same share of its size, adding up
# to that many pops (the jobs they leave are refilled by the unemployed): share = unemployed * attrition / population.
lfg_penal_idle_hands_attrition = {
	export_trigger_value_to_variable = {
		trigger = num_unemployed
		variable = lfg_attrition_share
	}
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = { always = yes }
		}
		variable = lfg_attrition_population
	}
	if = {
		limit = {
			check_variable = { which = lfg_attrition_share value > 0 }
			check_variable = { which = lfg_attrition_population value > 0 }
		}
		multiply_variable = { which = lfg_attrition_share value = @lfg_idle_hands_attrition }
		divide_variable = { which = lfg_attrition_share value = lfg_attrition_population }
		every_owned_pop_group = {
			export_trigger_value_to_variable = {
				trigger = pop_group_size
				variable = lfg_attrition_kill
			}
			multiply_variable = { which = lfg_attrition_kill value = prev.lfg_attrition_share }
			floor_variable = lfg_attrition_kill
			if = {
				limit = { check_variable = { which = lfg_attrition_kill value > 0 } }
				kill_pop_group = {
					pop_group = this
					amount = lfg_attrition_kill
				}
			}
		}
	}
}
