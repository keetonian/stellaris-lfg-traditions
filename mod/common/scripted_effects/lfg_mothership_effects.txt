# LFG Traditions - Mothership swarm (events/lfg_mothership_events.txt). Scope: the Mothership's fleet (ROOT there).

# Grow reserve $RESERVE$ (fleet variable) by $AMOUNT$, so that reserve + ships of size $SIZE$ in the escort stay within
# $CAP$. Nothing without $TRADITION$, the Nanotech tradition that gives designs for the size.
lfg_mothership_grow_reserve = {
	if = {
		limit = { owner = { has_tradition = $TRADITION$ } }
		change_variable = { which = $RESERVE$ value = $AMOUNT$ }
		set_variable = { which = lfg_reserve_room value = $CAP$ }
		owner = {
			every_owned_fleet = {
				limit = { has_fleet_flag = lfg_escort_of_@ROOT }
				every_owned_ship = {
					limit = { is_ship_size = $SIZE$ }
					ROOT = { change_variable = { which = lfg_reserve_room value = -1 } }
				}
			}
		}
		if = {
			limit = { check_variable = { which = $RESERVE$ value > lfg_reserve_room } }
			set_variable = { which = $RESERVE$ value = lfg_reserve_room }
		}
		if = {
			limit = { check_variable = { which = $RESERVE$ value < 0 } }
			set_variable = { which = $RESERVE$ value = 0 }
		}
	}
}

# Scope: the Mothership (ship; ROOT = its fleet). Adds its summed $MODIFIER$ (swarm modules) to ROOT's variable $TO$.
lfg_mothership_add_module = {
	export_modifier_to_variable = { modifier = $MODIFIER$ variable = lfg_module_value }
	ROOT = { change_variable = { which = $TO$ value = prev.lfg_module_value } }
	clear_variable = lfg_module_value
}

# Launch the whole reserve into the escort fleet (created beside the Mothership if it isn't out yet).
lfg_mothership_deploy = {
	save_event_target_as = lfg_mothership_fleet
	owner = { save_event_target_as = lfg_escort_owner }
	if = {
		limit = { NOT = { owner = { any_owned_fleet = { has_fleet_flag = lfg_escort_of_@ROOT } } } }
		create_fleet = {
			name = "lfg_mothership_escort_fleet"
			settings = {
				spawn_debris = no
				uses_naval_capacity = no
				can_upgrade = no
				can_change_leader = no
			}
			effect = {
				set_owner = event_target:lfg_escort_owner
				set_fleet_flag = lfg_mothership_escort
				set_fleet_flag = lfg_escort_of_@ROOT
				set_location = event_target:lfg_mothership_fleet
				set_fleet_stance = aggressive
				lfg_mothership_launch_reserve = yes
			}
		}
	}
	else = {
		owner = {
			every_owned_fleet = {
				limit = { has_fleet_flag = lfg_escort_of_@ROOT }
				lfg_mothership_launch_reserve = yes
			}
		}
	}
	owner = {
		every_owned_fleet = {
			limit = { has_fleet_flag = lfg_escort_of_@ROOT }
			auto_follow_fleet = { target = event_target:lfg_mothership_fleet }
		}
	}
}

# Scope: the escort fleet (event_target:lfg_mothership_fleet = its Mothership). Moves the reserve into it, one ship at a
# time (random picks from the owner's designs).
lfg_mothership_launch_reserve = {
	while = {
		limit = { event_target:lfg_mothership_fleet = { check_variable = { which = lfg_swarmers value > 0 } } }
		create_ship = {
			name = random
			random_existing_design = nanite_swarmer
		}
		event_target:lfg_mothership_fleet = { change_variable = { which = lfg_swarmers value = -1 } }
	}
	while = {
		limit = { event_target:lfg_mothership_fleet = { check_variable = { which = lfg_interdictors value > 0 } } }
		create_ship = {
			name = random
			random_existing_design = nanite_interdictor
		}
		event_target:lfg_mothership_fleet = { change_variable = { which = lfg_interdictors value = -1 } }
	}
}

# Recall the swarm: its surviving Swarmers and Interdictors go back to the reserve, and the escort fleet vanishes.
lfg_mothership_recall = {
	remove_fleet_flag = lfg_mothership_deployed
	owner = {
		every_owned_fleet = {
			limit = { has_fleet_flag = lfg_escort_of_@ROOT }
			every_owned_ship = {
				limit = { is_ship_size = nanite_swarmer }
				ROOT = { change_variable = { which = lfg_swarmers value = 1 } }
			}
			every_owned_ship = {
				limit = { is_ship_size = nanite_interdictor }
				ROOT = { change_variable = { which = lfg_interdictors value = 1 } }
			}
			delete_fleet = this
		}
	}
}

# System scope, from = the Progenitor Forge's builder (its site's on_build_complete): every world Gigastructural lets be
# dismantled for bulk matter (can_dismantle_planet_for_bm: not the star, colonies, habitats, megastructure anchors, holy
# worlds...) becomes bulk matter for the builder, 1 per size (Gigastructural's rate), and is removed; so are the system's
# debris and asteroid belts. lfg_mothership.10 tells the builder how much.
lfg_progenitor_forge_consume_system = {
	from = { set_variable = { which = lfg_forge_bulk_matter value = 0 } }
	every_system_planet = {
		limit = { can_dismantle_planet_for_bm = yes }
		export_trigger_value_to_variable = { trigger = planet_size variable = lfg_forge_size }
		from = { change_variable = { which = lfg_forge_bulk_matter value = prev.lfg_forge_size } }
		remove_planet = yes
	}
	giga_remove_system_debris = yes
	set_asteroid_belt = { radius = 0 }
	from = {
		add_resource = {
			giga_sr_bulk_matter = 1
			mult = lfg_forge_bulk_matter
		}
		country_event = { id = lfg_mothership.10 }
	}
}
