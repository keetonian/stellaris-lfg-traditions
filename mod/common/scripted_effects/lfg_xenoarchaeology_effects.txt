# LFG Traditions - Xenoarchaeology: Commissioned Digs spawns (options of lfg_xenoarchaeology.2). Scope: country.
# Each creates its thing within our borders, saves it as event_target:lfg_dig_location, sets the matching
# lfg_dig_<type> flag and shows lfg_xenoarchaeology.3; if no spot fits after all, it refunds the cost instead.

lfg_commission_dig_site = {
	lfg_commission_dig_reset = yes
	random_planet_within_border = {
		limit = { can_spawn_random_archaeological_site = yes }
		create_archaeological_site = random
		save_event_target_as = lfg_dig_location
		root = { set_country_flag = lfg_dig_site }
	}
	lfg_commission_dig_finish = yes
}

# spawn_random_anomaly needs a surveyor: any of our science ships (vanilla deep_scan_success_effect does the same).
lfg_commission_dig_anomaly = {
	lfg_commission_dig_reset = yes
	random_owned_ship = {
		limit = { is_ship_class = shipclass_science_ship }
		save_event_target_as = lfg_dig_surveyor
	}
	if = {
		limit = { exists = event_target:lfg_dig_surveyor }
		random_planet_within_border = {
			limit = {
				lfg_commission_anomaly_planet = yes
				can_spawn_random_anomaly = event_target:lfg_dig_surveyor
			}
			spawn_random_anomaly = {
				target = root
				surveyor = event_target:lfg_dig_surveyor
			}
			save_event_target_as = lfg_dig_location
			root = { set_country_flag = lfg_dig_anomaly }
		}
	}
	lfg_commission_dig_finish = yes
}

# Same spawn as vanilla's astral_splitting (astral_planes_effects.txt), without its weights.
lfg_commission_dig_rift = {
	lfg_commission_dig_reset = yes
	random_system_within_border = {
		limit = { can_spawn_astral_rift = yes }
		spawn_astral_rift = {
			random_pos = yes
			orbit_angle = 360
			spawn_sound = no
		}
		save_event_target_as = lfg_dig_location
		root = { set_country_flag = lfg_dig_rift }
	}
	lfg_commission_dig_finish = yes
}

lfg_commission_dig_reset = {
	remove_country_flag = lfg_dig_site
	remove_country_flag = lfg_dig_anomaly
	remove_country_flag = lfg_dig_rift
}

lfg_commission_dig_finish = {
	if = {
		limit = {
			OR = {
				has_country_flag = lfg_dig_site
				has_country_flag = lfg_dig_anomaly
				has_country_flag = lfg_dig_rift
			}
		}
		country_event = { id = lfg_xenoarchaeology.3 }
	}
	else = {
		add_resource = { minor_artifacts = @lfg_commission_dig_cost }
	}
}
