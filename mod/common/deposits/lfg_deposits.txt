# Sealed Ring origin: Leviathan Residue sealing a section of an intact ring world segment (added by lfg_sealed_ring.1: 3 on the
# homeworld, 7 on each other segment): -1 district, -10% habitability, +2 food. Clearing one gives 150 alloys and 10% progress in a random new research option;
# clearing the last offers the way back to the galaxy (lfg_sealed_ring.2).
d_lfg_sealed_section = {
	is_for_colonizable = yes
	icon = "d_mining_tunnels"
	category = deposit_cat_blockers
	time = 360

	potential = { always = no }

	planet_modifier = {
		planet_max_districts_add = -1
		pop_environment_tolerance = -0.1
	}

	# Food from what the Matriarch left behind (a blocker can produce and cost, like vanilla's nanite blockers).
	resources = {
		category = deposit_blockers
		produces = {
			food = 2
		}
		cost = {
			energy = 1000
		}
	}

	on_cleared = {
		owner = {
			add_resource = { alloys = 150 }
			# The tooltip only promises "a random research option": the game would otherwise preview the very techs.
			custom_tooltip = d_lfg_sealed_section_research_tt
			hidden_effect = {
				random_list = {
					1 = { add_random_research_option = { area = physics add_progress = 0.1 } }
					1 = { add_random_research_option = { area = society add_progress = 0.1 } }
					1 = { add_random_research_option = { area = engineering add_progress = 0.1 } }
				}
				# A day later, once this deposit is gone: were all of them cleared?
				country_event = { id = lfg_sealed_ring.2 days = 1 }
			}
		}
	}

	ai_weight = {
		weight = 5
	}

	drop_weight = { weight = 0 }
}
