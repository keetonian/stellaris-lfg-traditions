			id = lfg_nanite_siege_driver_debuff
			priority = 1
		}
	}
	on_hit = {
		hidden_effect = {
			solar_system = {
				every_ship_in_system = {
					limit = {
						NOT = { is_same_value = root }
						controller = { is_hostile = from.controller }
						distance = {
							source = root
							max_distance <= 45
							same_solar_system = yes
						}
					}
					set_variable_to_random_value = {
						which = lfg_nanite_burst_damage

						max = @lfg_nanite_siege_driver_burst_max

					}
					damage_ship = {
						amount = lfg_nanite_burst_damage
						attacker = from
					}
					clear_variable = lfg_nanite_burst_damage
				}
			}
		}
		custom_tooltip = lfg_nanite_siege_driver_on_hit_tt
