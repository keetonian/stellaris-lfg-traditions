# Mothership (ap_lfg_mothership): stronger than Gigastructural's systemcraft (8x its hull, 4x its armour, its shields), on its
# sections (generated) with the mid section's guns turned into hangars: 45 in all, for the Nanite Swarm hangar
# (LFG_NANITE_SWARM_HANGAR) or any other. Giga's systemcraft components fit (generated trigger override). Its look: the
# systemcraft's worlds in nanite skins, in a swarm. Paid in nanites, built only at a Progenitor Forge. Each month its
# escort fleet (following it, no naval capacity) gains Nanite Swarmers and Interdictors (events/lfg_mothership_events.txt).
lfg_mothership = {
	graphical_culture = no
	entity = "lfg_mothership_entity"
	max_speed = 100
	acceleration = 0.10
	rotation_speed = 0.05
	collision_radius = 512
	formation_priority = 512
	base_buildtime = 3600
	modifier = {
		ship_evasion_mult = -1
		ship_armor_add = 24200000
		ship_shield_add = 3000000
		ship_speed_mult = 2
		ship_ftl_jumpdrive_range_mult = 1	# 2x jump drive range (not on the Steam page)
	}
	cloaking_forbidden = yes
	max_hitpoints = 100000000
	size_multiplier = 256
	fleet_slot_size = 1000
	section_slots = {
		"bowgun" =			{ locator = "root" }
		"bow" =				{ locator = "root" }
		"mid" =				{ locator = "root" }
		"stern" =			{ locator = "root" }
		"sternthrusters" =	{ locator = "root" }
		"kugel" =			{ locator = "root" }
	}
	num_target_locators = 0
	is_space_station = no
	icon = ship_size_atk_system
	map_counter_icon = ship_counter_500
	combat_disengage_chance = 0
	prerequisites = { "tech_lfg_mothership" }
	enable_default_design = yes
	is_designable = yes
	class = shipclass_military
	can_disable = no
	can_have_federation_design = no
	potential_construction = { lfg_is_progenitor_forge = yes }
	# One per empire, checked at the shipyard (this = the Progenitor Forge) so designing stays open: possible_country
	# also blocked the ship designer. Two queued at once slip past this; lfg_mothership.6 (on_ship_built) refunds the
	# extra one.
	possible_construction = {
		lfg_is_progenitor_forge = yes
		custom_tooltip = {
			fail_text = lfg_mothership_limit
			owner = { NOT = { any_owned_ship = { is_ship_size = lfg_mothership } } }
		}
		# Its crew: 2,000 pops, taken by the decision_lfg_mothership_crew planet decision (like Gigastructural's
		# systemcraft crew); used up when the Mothership is finished (lfg_mothership.13).
		custom_tooltip = {
			fail_text = lfg_requires_mothership_crew
			owner = { has_country_flag = lfg_mothership_crew_ready }
		}
	}
	construction_type = starbase_shipyard

	ship_roles = { carrier artillery }

	required_component_set = "power_core"
	required_component_set = "ftl_components"
	required_component_set = "sensor_components"
	required_component_set = "combat_computers"
	required_component_set = "ship_aura_components"

	# Cost and upkeep are the hull's only (components add nothing, like vanilla's Nanite Swarmers).
	components_add_to_cost = no
	resources = {
		category = ships
		cost = {
			nanites = @lfg_mothership_cost
			alloys = @lfg_mothership_alloy_cost
			influence = @lfg_mothership_influence_cost
			giga_sr_bulk_matter = @lfg_mothership_bulk_matter_cost
		}
		upkeep = {
			nanites = @lfg_mothership_upkeep
		}
		logistics = {
			trade = 500
		}
	}
}
