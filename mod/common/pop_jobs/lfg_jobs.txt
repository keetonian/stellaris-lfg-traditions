# LFG Traditions - jobs.

# Penitence: inmates untangle supertensiles by hand (modelled on Gigastructural's giga_supertensiles_extractor and
# vanilla's foundry_prison_worker). Provided only by zone_lfg_penal_supertensiles, which needs supertensiles
# (global flag giga_buildcap_j), a penal colony and Prison Industry.
lfg_penal_supertensile_laborer = {
	category = worker
	swappable_data = {
		default = {
			condition_string = WORKER_JOB_TRIGGER
			building_icon = building_foundry_1
			icon = giga_supertensiles_extractor
		}
	}

	tags = { giga_supertensiles }

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_worker_job

	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
	}

	resources = {
		category = planet_lfg_penal_laborers
		produces = {
			giga_sr_amb_megaconstruction = @lfg_penal_supertensile_output
		}
		upkeep = {
			minerals = @lfg_penal_supertensile_mineral_upkeep
			alloys = @lfg_penal_supertensile_alloys_upkeep
			unity = @lfg_penal_supertensile_unity_upkeep
		}
	}

	weight = {
		weight = @specialist_job_weight
		modifier = {
			factor = 2
			is_enslaved = yes
			can_take_servant_job = no
		}
	}
}
