# LFG Traditions - general scripted triggers.

# Scope: country. The empire's main species is a food-eating (organic) species: vanilla gives food upkeep through the
# hidden trait_organic (biological species; phototrophic ones still eat, just less), see
# common/traits/02_species_traits_basic_characteristics.txt. Voidlings count too (by design). Lithoids eat minerals;
# machines and robots don't eat.
lfg_main_species_eats_food = {
	owner_species = {
		has_trait = trait_organic
	}
}

# Storm Sovereignty. Country scope: Storm Chasers who took the perk ignore cosmic storms' penalties.
lfg_storm_immune_country = {
	has_origin = origin_storm_chasers
	has_ascension_perk = ap_lfg_storm_sovereignty
}

# Country scope: owns a finished Tempest Lodestar and has Storm Sovereignty (bonuses inside nexus storms). A Lodestar
# captured by an empire without the perk does nothing for it.
lfg_has_tempest_lodestar = {
	has_megastructure = lfg_tempest_lodestar_1
	has_ascension_perk = ap_lfg_storm_sovereignty
}

# Country scope: gets Storm Sovereignty's nexus storm bonuses: a Tempest Lodestar's owner, or Storm Chasers with the perk.
lfg_nexus_storm_bonuses = {
	OR = {
		lfg_has_tempest_lodestar = yes
		lfg_storm_immune_country = yes
	}
}

# Here Be Dragons. Ship scope: an Elder Dragon that isn't the Legendary one (the ship size of growth step 11).
lfg_is_elder_drake = {
	is_ship_size = lfg_drake_11
	NOT = { has_ship_flag = lfg_legendary_dragon }
}

# Country scope: owns the Legendary Dragon (only one at a time; ap_lfg_legendary_dragons).
lfg_has_legendary_dragon = {
	any_owned_ship = { has_ship_flag = lfg_legendary_dragon }
}

# Here Be Dragons. Ship scope (a drake): can still grow (below the Elder, or the Legendary Dragon below step 24).
lfg_drake_can_grow = {
	OR = {
		check_variable = { which = lfg_drake_step value < 11 }
		AND = {
			has_ship_flag = lfg_legendary_dragon
			check_variable = { which = lfg_drake_step value < 24 }
		}
	}
}

# Ship scope: a leviathan whose death feeds a drake's growth (Legendary Dragons' hunting; lfg_dragons.7): the killable
# organic ones - Dimensional Horror (also the Elder One and the Worm's avatar), Stellar Devourer, Corrupted Avatar,
# Tiyanki Matriarch, Voidspawn, Toxic God, Ether Drake, Shard and the Sky Dragon.
lfg_drake_prey = {
	OR = {
		is_ship_size = dimensional_horror
		is_ship_size = stellarite
		is_ship_size = corrupted_avatar
		is_ship_size = leviathan_01_elder_tiyanki
		is_ship_size = leviathan_01_voidspawn
		is_ship_size = toxic_god
		is_ship_size = space_dragon_red
		is_ship_size = space_dragon_blue
		is_ship_size = space_dragon_origin
	}
}

# Country scope: may feed its own pops to a drake (Tribute): not xenophiles, hives or virtual machines.
lfg_can_feed_pops_to_dragons = {
	is_xenophile = no
	is_hive_empire = no
	NOT = { has_tradition = tr_virtuality_adopt }
}
